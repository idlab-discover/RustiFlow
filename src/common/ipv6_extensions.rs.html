1/// Extension headers whose layout must be handled before transport parsing.
2#[derive(Clone, Copy)]
3pub enum Ipv6Extension {
4    Options,
5    Fragment,
6    Authentication,
7    Encrypted,
8}
9
10impl Ipv6Extension {
11    #[inline(always)]
12    pub fn from_next_header(protocol: u8) -> Option<Self> {
13        match protocol {
14            0 | 43 | 60 | 135 | 139 | 140 => Some(Self::Options),
15            44 => Some(Self::Fragment),
16            51 => Some(Self::Authentication),
17            50 => Some(Self::Encrypted),
18            _ => None,
19        }
20    }
21
22    /// Returns the length to skip, or rejects encrypted/non-initial fragments.
23    /// The caller must also check that the complete header is available.
24    #[inline(always)]
25    pub fn header_length(self, prefix: [u8; 4]) -> Option<usize> {
26        match self {
27            Self::Options => Some((usize::from(prefix[1]) + 1) * 8),
28            Self::Fragment => {
29                let offset_and_flags = u16::from_be_bytes([prefix[2], prefix[3]]);
30                (offset_and_flags & 0xfff8 == 0).then_some(8)
31            }
32            Self::Authentication => Some((usize::from(prefix[1]) + 2) * 4),
33            Self::Encrypted => None,
34        }
35    }
36}

38#[cfg(test)]
39mod tests {
40    use super::Ipv6Extension;
41
42    #[test]
43    fn extension_lengths_and_fragment_offsets_follow_the_wire_format() {
44        for (protocol, prefix, expected) in [
45            (0, [6, 1, 0, 0], Some(16)),
46            (44, [17, 0, 0, 1], Some(8)),
47            (44, [17, 0, 0, 8], None),
48            (44, [17, 0, 255, 249], None),
49            (51, [6, 1, 0, 0], Some(12)),
50            (50, [6, 0, 0, 0], None),
51        ] {
52            assert_eq!(
53                Ipv6Extension::from_next_header(protocol)
54                    .unwrap()
55                    .header_length(prefix),
56                expected,
57                "protocol {protocol}"
58            );
59        }
60        assert!(Ipv6Extension::from_next_header(6).is_none());
61    }
