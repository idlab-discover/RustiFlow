8/// Expects a result slot for each packet. Only call on CPUs with AVX2.
9#[target_feature(enable = "avx2")]
10pub(super) fn orientations(packets: &[PacketFeatures], output: &mut [KeyOrientation]) {
11    let (pairs, tail) = packets.as_chunks::<2>();
12    let (result_pairs, result_tail) = output.as_chunks_mut::<2>();
13    for ([first, second], result) in pairs.iter().zip(result_pairs) {
14        if let (Some((first_src, first_dst)), Some((second_src, second_dst))) =
15            (addresses(first), addresses(second))
16        {
17            // SAFETY: these initialized bytes fill the vector, one address
18            // per 128-bit lane, and every bit pattern is valid for __m256i.
19            let src =
20                unsafe { std::mem::transmute::<[[u8; 16]; 2], __m256i>([first_src, second_src]) };
21            let dst =
22                unsafe { std::mem::transmute::<[[u8; 16]; 2], __m256i>([first_dst, second_dst]) };
23
24            let differences = !_mm256_movemask_epi8(_mm256_cmpeq_epi8(src, dst)) as u32;
25            let src_greater =
26                !_mm256_movemask_epi8(_mm256_cmpeq_epi8(src, _mm256_min_epu8(src, dst))) as u32;
27            result[0] = orientation(first, differences as u16, src_greater as u16);
28            result[1] = orientation(
29                second,
30                (differences >> 16) as u16,
31                (src_greater >> 16) as u16,
32            );
33        } else {
34            result[0] = first.flow_key_value().canonicalize().1;
35            result[1] = second.flow_key_value().canonicalize().1;
36        }
37    }
38
39    if let [packet] = tail {
40        result_tail[0] = packet.flow_key_value().canonicalize().1;
41    }
42}

44fn addresses(packet: &PacketFeatures) -> Option<([u8; 16], [u8; 16])> {
45    match (packet.source_ip, packet.destination_ip) {
46        (IpAddr::V4(src), IpAddr::V4(dst)) => Some((
47            u128::from(src.to_bits()).to_be_bytes(),
48            u128::from(dst.to_bits()).to_be_bytes(),
49        )),
50        (IpAddr::V6(src), IpAddr::V6(dst)) => Some((src.octets(), dst.octets())),
51        (IpAddr::V4(_), IpAddr::V6(_)) | (IpAddr::V6(_), IpAddr::V4(_)) => None,
52    }
53}
54
55fn orientation(packet: &PacketFeatures, differences: u16, src_greater: u16) -> KeyOrientation {
56    let reversed = if differences == 0 {
57        packet.source_port > packet.destination_port
58    } else {
59        // Address octet zero is the least significant mask bit.
60        src_greater & (1 << differences.trailing_zeros()) != 0
61    };
62    if reversed {
63        KeyOrientation::Reversed
64    } else {
65        KeyOrientation::Canonical
66    }
67}

69#[cfg(test)]
70mod tests {
71    use std::net::{Ipv4Addr, Ipv6Addr};
72
73    use super::*;
74
75    #[test]
76    fn avx2_orientations_match_scalar_in_packet_order() {
77        if !std::is_x86_feature_detected!("avx2") {
78            return;
79        }
80
81        let mut packets = Vec::new();
82        let mut add_both_directions = |src, dst, src_port, dst_port| {
83            for (source_ip, destination_ip, source_port, destination_port) in [
84                (src, dst, src_port, dst_port),
85                (dst, src, dst_port, src_port),
86            ] {
87                packets.push(PacketFeatures {
88                    source_ip,
89                    destination_ip,
90                    source_port,
91                    destination_port,
92                    ..PacketFeatures::default()
93                });
94            }
95        };
96
97        // Exercise each possible first differing IPv6 octet, an opposing
98        // later difference, unsigned ordering and the address-equality tie.
99        let base = [127; 16];
100        let v6 = IpAddr::from(Ipv6Addr::from(base));
101        for index in 0..16 {
102            let mut other = base;
103            other[15] = 0;
104            other[index] = 128;
105            add_both_directions(v6, Ipv6Addr::from(other).into(), 65535, 0);
106        }
107        let v4 = IpAddr::from(Ipv4Addr::new(127, 255, 0, 1));
108        add_both_directions(v4, Ipv4Addr::new(128, 0, 0, 0).into(), 65535, 0);
109        add_both_directions(v4, v4, 65535, 0);
110        add_both_directions(v6, v6, 0, 65535);
111        add_both_directions(v6, v6, 443, 443);
112        add_both_directions(v4, v6, 65535, 0);
113        packets.push(PacketFeatures::default());
114
115        for input in [packets.as_slice(), &packets[1..], &packets[..1], &[]] {
116            let expected: Vec<_> = input
117                .iter()
118                .map(|packet| packet.flow_key_value().canonicalize().1)
119                .collect();
120            let mut actual = vec![KeyOrientation::Reversed; input.len()];
121            // SAFETY: AVX2 support was checked above.
122            unsafe { orientations(input, &mut actual) };
123            assert_eq!(actual, expected);
124        }
125    }
