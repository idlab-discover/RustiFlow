1//! Load-time overrides for the existing event maps. Map names are ELF symbols,
2//! so the classifier and userspace event ABIs do not depend on ring capacity.
3use anyhow::{ensure, Context};
4use common::REALTIME_EVENT_QUEUE_COUNT;
5use log::info;
6
7const SETTING: &str = "RUSTIFLOW_EXPERIMENT_RING_BYTES";
8
9pub(super) fn map_overrides(ingress_only: bool) -> anyhow::Result<Vec<(String, u32)>> {
10    let Some(value) = std::env::var_os(SETTING) else {
11        return Ok(Vec::new());
12    };
13    // SAFETY: sysconf with this constant takes no pointers and has no side effects.
14    let page = u32::try_from(unsafe { libc::sysconf(libc::_SC_PAGESIZE) })
15        .context("reading host page size for event rings")?;
16    ensure!(page.is_power_of_two(), "invalid host page size {page}");
17    let directions = if ingress_only { 1 } else { 2 };
18    let (v4, v6, total) = parse_bytes(
19        value
20            .to_str()
21            .with_context(|| format!("non-UTF-8 {SETTING}"))?,
22        page,
23        directions,
24    )
25    .with_context(|| format!("invalid {SETTING}"))?;
26    info!("Event ring bytes per map: IPv4={v4}, IPv6={v6}, total data capacity={total}");
27    Ok([("EVENTS_IPV4", v4), ("EVENTS_IPV6", v6)]
28        .into_iter()
29        .flat_map(|(prefix, bytes)| {
30            (0..REALTIME_EVENT_QUEUE_COUNT).map(move |index| (format!("{prefix}_{index}"), bytes))
31        })
32        .collect())
33}
34
35fn parse_bytes(value: &str, page: u32, directions: u64) -> anyhow::Result<(u32, u32, u64)> {
36    let syntax = "expected IPv4_BYTES,IPv6_BYTES,TOTAL_CAP_BYTES";
37    let (v4, rest) = value.split_once(',').context(syntax)?;
38    let (v6, cap) = rest.split_once(',').context(syntax)?;
39    let cap = cap
40        .parse::<u64>()
41        .context("expected a decimal total ring data cap")?;
42    let parse = |s: &str| -> anyhow::Result<u32> {
43        let bytes = s
44            .parse::<u32>()
45            .context("expected a decimal u32 byte count")?;
46        ensure!(
47            bytes >= page && bytes.is_power_of_two(),
48            "ring size {bytes} must be a power of two of at least {page} bytes"
49        );
50        Ok(bytes)
51    };
52    let (v4, v6) = (parse(v4)?, parse(v6)?);
53    let total = (u64::from(v4) + u64::from(v6)) * REALTIME_EVENT_QUEUE_COUNT as u64 * directions;
54    ensure!(
55        total <= cap,
56        "{directions} capture direction(s) need {total} ring data bytes, cap is {cap}"
57    );
58    Ok((v4, v6, total))
59}

61#[cfg(test)]
62mod tests {
63    use super::parse_bytes;
64
65    #[test]
66    fn budget_includes_every_family_queue_and_capture_direction() {
67        let ingress = "8388608,131072,68157440";
68        assert_eq!(
69            parse_bytes(ingress, 65536, 1).unwrap(),
70            (8388608, 131072, 68157440)
71        );
72        assert!(parse_bytes(ingress, 65536, 2).is_err());
73        assert_eq!(
74            parse_bytes("8388608,131072,136314880", 65536, 2).unwrap(),
75            (8388608, 131072, 136314880)
76        );
77    }
78
79    #[test]
80    fn invalid_settings_reject_rounding_and_overspend() {
81        for bad in [
82            "",
83            "4096",
84            "4096,4096",
85            "4096,4096,65535",
86            "4096,0,65536",
87            "12288,4096,999999",
88            "4294967296,4096,999999",
89            "4096,4096,65536,1",
90        ] {
91            assert!(parse_bytes(bad, 4096, 1).is_err(), "{bad}");
92        }
93        assert!(parse_bytes("65536,4096,999999", 65536, 1).is_err());
94    }
