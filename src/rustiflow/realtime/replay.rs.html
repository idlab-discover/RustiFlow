1//! Synthetic event replay through the realtime userspace pipeline.
2//!
3//! Events are generated in memory before timing starts, then drained through
4//! the same batching, dispatch, shard and export stages as live capture. No
5//! eBPF program is loaded, so the result is the userspace ceiling independent
6//! of any traffic generator.

34const IPV4_HEADER_BYTES: u16 = 20;
35const IPV6_HEADER_BYTES: u16 = 40;
36const TCP_HEADER_BYTES: u8 = 20;
37const UDP_HEADER_BYTES: u8 = 8;
38const TCP_ACK_PSH: u8 = 0x18;
39const TCP_WINDOW: u16 = 65_535;
40const CLIENT_PORT: u16 = 40_000;
41const TCP_SERVER_PORT: u16 = 443;
42const UDP_SERVER_PORT: u16 = 4_433;
43const REPLAY_SEED: u64 = 0x5eed_f10e;

45/// Generates the events, replays them and prints throughput and CPU cost.
46///
47/// The measured interval runs from the first replayed event until the output
48/// writer has closed, so it includes the final export of every open flow.
49pub async fn replay_bench<T: Flow>(
50    replay_args: &ReplayBenchArgs,
51    config: &ExportConfig,
52    num_threads: u8,
53    exporter: impl FlowSink<T>,
54    output_task: JoinHandle<()>,
55) -> Result<(), anyhow::Error> {
56    let shard_count =
57        NonZeroUsize::new(usize::from(num_threads)).context("thread count must be at least 1")?;
58    let realtime_offset_us = compute_realtime_offset_us()?;
59    let generation_start = Instant::now();
60    let events = ReplayEvents::generate(replay_args)?;
61    eprintln!(
62        "Generated {} {} events for {} flows over {} sources in {:.2} s",
63        replay_args.packets,
64        events.family_label(),
65        replay_args.flows,
66        replay_args.sources,
67        generation_start.elapsed().as_secs_f64(),
68    );
69
70    let profiling_session = ProfilingSession::start_from_env("replay").unwrap_or_else(|err| {
71        error!("Error starting profiler: {:?}", err);
72        None
73    });
74    let cpu_start = process_cpu_time()?;
75    let wall_start = Instant::now();
76
77    let settings = PipelineSettings::from_env();
78    let enable_source_stats = std::env::var_os("RUSTIFLOW_REALTIME_STATS").is_some();
79    let shard_selector = ShardSelector::new(shard_count);
80    let shard_senders = spawn_shard_workers(
81        shard_count,
82        settings.shard_queue_capacity,
83        &FlowTableSettings {
84            active_timeout: config.active_timeout,
85            idle_timeout: config.idle_timeout,
86            early_export: config.early_export,
87            expiration_check_interval: config.expiration_check_interval,
88        },
89        &exporter,
90    );
91    // The shard encoders hold their own chunk senders; the writer closes once
92    // they finish.
93    drop(exporter);
94
95    let mut handle_set = JoinSet::new();
96    let mut source_stats = Vec::new();
97    let mut new_batcher = |handle_set: &mut JoinSet<()>, label: String| {
98        let stats = enable_source_stats.then(|| Arc::new(RealtimeSourceStats::default()));
99        source_stats.push((label, stats.clone()));
100        SourceBatcher::new(
101            handle_set,
102            &shard_senders,
103            &shard_selector,
104            &settings,
105            stats,
106        )
107    };
108    match events {
109        ReplayEvents::Ipv4(sources) => {
110            for (index, events) in sources.into_iter().enumerate() {
111                let batcher = new_batcher(&mut handle_set, queue_label("replay-ipv4", index));
112                handle_set.spawn(replay_source(
113                    events,
114                    batcher,
115                    replay_args.drain_burst,
116                    PacketFeatures::from_ebpf_event_ipv4,
117                    realtime_offset_us,
118                ));
119            }
120        }
121        ReplayEvents::Ipv6(sources) => {
122            for (index, events) in sources.into_iter().enumerate() {
123                let batcher = new_batcher(&mut handle_set, queue_label("replay-ipv6", index));
124                handle_set.spawn(replay_source(
125                    events,
126                    batcher,
127                    replay_args.drain_burst,
128                    PacketFeatures::from_ebpf_event_ipv6,
129                    realtime_offset_us,
130                ));
131            }
132        }
133    }
134    // Only the dispatchers keep the shards' senders, so the shards finish once
135    // every source has drained.
136    drop(shard_senders);
137
138    while let Some(result) = handle_set.join_next().await {
139        if let Err(err) = result {
140            error!("Replay task failed: {:?}", err);
141        }
142    }
143    output_task.await.context("output writer task failed")?;
144
145    let wall = wall_start.elapsed();
146    let cpu = process_cpu_time()?.saturating_sub(cpu_start);
147
148    if let Some(profiling_session) = profiling_session {
149        if let Err(err) = profiling_session.finish() {
150            error!("Error finishing profiler: {:?}", err);
151        }
152    }
153    export_profile::log_summary("replay");
154    for (label, stats) in &source_stats {
155        if let Some(stats) = stats {
156            log_source_stats(label, stats);
157        }
158    }
159
160    let packets = replay_args.packets.get() as f64;
161    eprintln!(
162        "Replay: packets={} threads={} wall_s={:.3} mpps={:.3} cpu_s={:.3} cpu_ns_per_packet={:.1} cpus_busy={:.2}",
163        replay_args.packets,
164        num_threads,
165        wall.as_secs_f64(),
166        packets / wall.as_secs_f64() / 1e6,
167        cpu.as_secs_f64(),
168        cpu.as_nanos() as f64 / packets,
169        cpu.as_secs_f64() / wall.as_secs_f64(),
170    );
171    Ok(())
172}

174/// Drains one source's events in bursts, flushing partial batches after each
175/// burst as a live source does when its ring runs empty.
176async fn replay_source<E: Send + Sync + 'static>(
177    events: Vec<E>,
178    mut batcher: SourceBatcher,
179    drain_burst: NonZeroUsize,
180    decode: fn(&E, i64) -> PacketFeatures,
181    realtime_offset_us: i64,
182) {
183    for burst in events.chunks(drain_burst.get()) {
184        for event in burst {
185            batcher
186                .push_event(|| decode(event, realtime_offset_us))
187                .await;
188        }
189        batcher.flush().await;
190        // A live source waits for ring readiness between drains.
191        tokio::task::yield_now().await;
192    }
193}

195/// Pre-generated events, one list per source. All events of a flow go to the
196/// same source, like the eBPF biflow queue selection, but sources are evenly
197/// balanced, so queue skew is not modelled.
198enum ReplayEvents {
199    Ipv4(Vec<Vec<EbpfEventIpv4>>),
200    Ipv6(Vec<Vec<EbpfEventIpv6>>),
201}

203/// One synthetic packet before it is encoded as an IPv4 or IPv6 event.
204struct SyntheticPacket {
205    timestamp_ns: u64,
206    flow: u32,
207    forward: bool,
208    is_tcp: bool,
209    sequence_number: u32,
210    sequence_number_ack: u32,
211}
212
213impl ReplayEvents {
214    /// Spreads packets uniformly at random over the flows. Even flows are TCP
215    /// with in-order ACK|PSH data segments in both directions; odd flows are
216    /// UDP. Flows never terminate, so they are exported at the end.
217    fn generate(replay_args: &ReplayBenchArgs) -> Result<Self, anyhow::Error> {
218        let source_count = replay_args.sources.get();
219        let packets = replay_args.packets.get();
220        let per_source = (packets as usize).div_ceil(source_count);
221        let start_ns = read_clock_us(libc::CLOCK_MONOTONIC)? as u64 * 1_000;
222        let virtual_pps = replay_args.virtual_pps.get();
223        let payload_bytes = replay_args.payload_bytes;
224
225        let mut rng = SplitMix64(REPLAY_SEED);
226        // Next sequence number per flow and direction.
227        let mut next_sequence: Vec<[u32; 2]> = (0..replay_args.flows)
228            .map(|flow| [flow.wrapping_mul(7_919), flow.wrapping_mul(104_729)])
229            .collect();
230        let synthetic_packets = (0..packets).map(|index| {
231            let flow = (rng.next() % u64::from(replay_args.flows)) as u32;
232            let forward = rng.next() & 1 == 0;
233            let is_tcp = flow.is_multiple_of(2);
234            let (sequence_number, sequence_number_ack) = if is_tcp {
235                let sequences = &mut next_sequence[flow as usize];
236                let (sending, receiving) = if forward { (0, 1) } else { (1, 0) };
237                let sequence_number = sequences[sending];
238                sequences[sending] = sequence_number.wrapping_add(u32::from(payload_bytes));
239                (sequence_number, sequences[receiving])
240            } else {
241                (0, 0)
242            };
243            SyntheticPacket {
244                timestamp_ns: start_ns
245                    + (u128::from(index) * 1_000_000_000 / u128::from(virtual_pps)) as u64,
246                flow,
247                forward,
248                is_tcp,
249                sequence_number,
250                sequence_number_ack,
251            }
252        });
253
254        if replay_args.ipv6 {
255            let mut sources = vec![Vec::with_capacity(per_source); source_count];
256            for packet in synthetic_packets {
257                let (client, server) = ipv6_endpoints(packet.flow);
258                sources[packet.flow as usize % source_count].push(encode_ipv6(
259                    &packet,
260                    client,
261                    server,
262                    payload_bytes,
263                ));
264            }
265            Ok(Self::Ipv6(sources))
266        } else {
267            let mut sources = vec![Vec::with_capacity(per_source); source_count];
268            for packet in synthetic_packets {
269                let (client, server) = ipv4_endpoints(packet.flow);
270                sources[packet.flow as usize % source_count].push(encode_ipv4(
271                    &packet,
272                    client,
273                    server,
274                    payload_bytes,
275                ));
276            }
277            Ok(Self::Ipv4(sources))
278        }
279    }
280
281    fn family_label(&self) -> &'static str {
282        match self {
283            Self::Ipv4(_) => "IPv4",
284            Self::Ipv6(_) => "IPv6",
285        }
286    }
287}

289fn ipv4_endpoints(flow: u32) -> (Ipv4Addr, Ipv4Addr) {
290    (
291        Ipv4Addr::from(u32::from(Ipv4Addr::new(10, 0, 0, 0)) | flow),
292        Ipv4Addr::new(192, 168, 0, 1),
293    )
294}
295
296fn ipv6_endpoints(flow: u32) -> (Ipv6Addr, Ipv6Addr) {
297    (
298        Ipv6Addr::from(u128::from(Ipv6Addr::new(0xfd00, 0, 0, 0, 0, 0, 0, 0)) | u128::from(flow)),
299        Ipv6Addr::new(0xfd01, 0, 0, 0, 0, 0, 0, 1),
300    )
301}

303/// Transport fields shared by both address families.
304struct TransportFields {
305    source_port: u16,
306    destination_port: u16,
307    header_bytes: u8,
308    flags: u8,
309    window_size: u16,
310    protocol: u8,
311}
312
313fn transport_fields(packet: &SyntheticPacket) -> TransportFields {
314    let (header_bytes, flags, window_size, protocol, server_port) = if packet.is_tcp {
315        (
316            TCP_HEADER_BYTES,
317            TCP_ACK_PSH,
318            TCP_WINDOW,
319            6,
320            TCP_SERVER_PORT,
321        )
322    } else {
323        (UDP_HEADER_BYTES, 0, 0, 17, UDP_SERVER_PORT)
324    };
325    let (source_port, destination_port) = if packet.forward {
326        (CLIENT_PORT, server_port)
327    } else {
328        (server_port, CLIENT_PORT)
329    };
330    TransportFields {
331        source_port,
332        destination_port,
333        header_bytes,
334        flags,
335        window_size,
336        protocol,
337    }
338}

340/// Encodes like the eBPF programs: addresses in network order, `length` as the
341/// IP total length, `header_length` as the transport header length.
342fn encode_ipv4(
343    packet: &SyntheticPacket,
344    client: Ipv4Addr,
345    server: Ipv4Addr,
346    payload_bytes: u16,
347) -> EbpfEventIpv4 {
348    let transport = transport_fields(packet);
349    let (source, destination) = if packet.forward {
350        (client, server)
351    } else {
352        (server, client)
353    };
354    EbpfEventIpv4::new(
355        packet.timestamp_ns,
356        u32::from(destination).to_be(),
357        u32::from(source).to_be(),
358        transport.destination_port,
359        transport.source_port,
360        payload_bytes,
361        IPV4_HEADER_BYTES + u16::from(transport.header_bytes) + payload_bytes,
362        transport.window_size,
363        transport.flags,
364        transport.protocol,
365        transport.header_bytes,
366        packet.sequence_number,
367        packet.sequence_number_ack,
368        0,
369        0,
370    )
371}

395        transport.protocol,
396        transport.header_bytes,
397        packet.sequence_number,
398        packet.sequence_number_ack,
399        0,
400        0,
401    )
402}

404/// Deterministic generator so runs replay identical traffic.
405struct SplitMix64(u64);
406
407impl SplitMix64 {
408    fn next(&mut self) -> u64 {
409        self.0 = self.0.wrapping_add(0x9e37_79b9_7f4a_7c15);
410        let mut z = self.0;
411        z = (z ^ (z >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
412        z = (z ^ (z >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
413        z ^ (z >> 31)
414    }
415}

417/// User plus system CPU time of the whole process.
418fn process_cpu_time() -> Result<Duration, anyhow::Error> {
419    let mut usage = std::mem::MaybeUninit::<libc::rusage>::uninit();
420    if unsafe { libc::getrusage(libc::RUSAGE_SELF, usage.as_mut_ptr()) } != 0 {
421        return Err(std::io::Error::last_os_error().into());
422    }
423    // SAFETY: getrusage succeeded and filled the struct.
424    let usage = unsafe { usage.assume_init() };
425    let timeval = |tv: libc::timeval| {
426        Duration::from_secs(tv.tv_sec as u64) + Duration::from_micros(tv.tv_usec as u64)
427    };
428    Ok(timeval(usage.ru_utime) + timeval(usage.ru_stime))
