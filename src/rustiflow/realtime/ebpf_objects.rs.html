1//! Locates the eBPF object files that realtime capture loads at runtime.
2//!
3//! A release install keeps the objects beside the executable's prefix in
4//! `<prefix>/lib/rustiflow/`, while development builds leave them in Cargo's
5//! target directory. Both layouts must work without rebuilding the binary.
6
7use std::path::{Path, PathBuf};
8
9use anyhow::bail;
10
11/// Overrides every other location. When set, only this directory is searched,
12/// so a typo fails loudly instead of silently loading a different build.
13pub const EBPF_DIR_ENV: &str = "RUSTIFLOW_EBPF_DIR";
14
15/// Where the release bundle installs the objects, relative to the prefix.
16const INSTALLED_SUBDIR: &str = "lib/rustiflow";
17
18/// eBPF target the userspace loader expects; the only supported one today.
19const EBPF_TARGET: &str = "bpfel-unknown-none";
20
21/// Resolves `program_name` to an existing object file from the current
22/// process environment, or reports every directory that was searched.
23pub fn ebpf_object_path(program_name: &str) -> Result<PathBuf, anyhow::Error> {
24    let search_dirs = ebpf_search_dirs(
25        std::env::var_os(EBPF_DIR_ENV).map(PathBuf::from),
26        std::env::current_exe().ok().as_deref(),
27        &development_target_dir(),
28    );
29
30    if let Some(path) = search_dirs
31        .iter()
32        .map(|dir| dir.join(program_name))
33        .find(|path| path.is_file())
34    {
35        return Ok(path);
36    }
37
38    let searched = search_dirs
39        .iter()
40        .map(|dir| format!("  {}", dir.display()))
41        .collect::<Vec<_>>()
42        .join("\n");
43    bail!(
44        "eBPF object {program_name} not found. Searched:\n{searched}\n\
45         Install a release bundle, set {EBPF_DIR_ENV}, or build it with `cargo xtask {}`.",
46        program_name.trim_start_matches("rustiflow-")
47    )
48}
49
50/// Directories to search, in priority order: the explicit override alone, or
51/// else the installed layout followed by the development target directory.
52fn ebpf_search_dirs(
53    override_dir: Option<PathBuf>,
54    executable: Option<&Path>,
55    development_target_dir: &Path,
56) -> Vec<PathBuf> {
57    if let Some(dir) = override_dir {
58        return vec![dir];
59    }
60
61    let installed = executable
62        .and_then(Path::parent)
63        .and_then(Path::parent)
64        .map(|prefix| prefix.join(INSTALLED_SUBDIR));
65    let profile = if cfg!(debug_assertions) {
66        "debug"
67    } else {
68        "release"
69    };
70    let development = development_target_dir.join(EBPF_TARGET).join(profile);
71
72    installed.into_iter().chain([development]).collect()
73}
74
75/// Cargo's target directory for `cargo xtask run` and other source-tree runs.
76fn development_target_dir() -> PathBuf {
77    std::env::var_os("CARGO_TARGET_DIR")
78        .map(PathBuf::from)
79        .unwrap_or_else(|| PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../target"))
80}

82#[cfg(test)]
83mod tests {
84    use super::*;
85
86    fn profile() -> &'static str {
87        if cfg!(debug_assertions) {
88            "debug"
89        } else {
90            "release"
91        }
92    }
93
94    #[test]
95    fn override_dir_is_the_only_candidate() {
96        let dirs = ebpf_search_dirs(
97            Some(PathBuf::from("/opt/ebpf")),
98            Some(Path::new("/usr/bin/rustiflow")),
99            Path::new("/src/target"),
100        );
101
102        assert_eq!(dirs, vec![PathBuf::from("/opt/ebpf")]);
103    }
104
105    #[test]
106    fn installed_layout_precedes_development_target() {
107        let dirs = ebpf_search_dirs(
108            None,
109            Some(Path::new("/usr/local/bin/rustiflow")),
110            Path::new("/src/target"),
111        );
112
113        assert_eq!(
114            dirs,
115            vec![
116                PathBuf::from("/usr/local/lib/rustiflow"),
117                PathBuf::from(format!("/src/target/bpfel-unknown-none/{}", profile())),
118            ]
119        );
120    }
121
122    #[test]
123    fn unknown_executable_falls_back_to_development_target() {
124        let dirs = ebpf_search_dirs(None, None, Path::new("/src/target"));
125
126        assert_eq!(
127            dirs,
128            vec![PathBuf::from(format!(
129                "/src/target/bpfel-unknown-none/{}",
130                profile()
131            ))]
132        );
133    }
134
135    #[test]
136    fn missing_objects_report_every_searched_dir() {
137        let error = ebpf_object_path("rustiflow-ebpf-does-not-exist")
138            .expect_err("object must not exist")
139            .to_string();
140
141        assert!(error.contains("rustiflow-ebpf-does-not-exist not found"));
142        assert!(error.contains("cargo xtask ebpf-does-not-exist"));
143    }
144}
