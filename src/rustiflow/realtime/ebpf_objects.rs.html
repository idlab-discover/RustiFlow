1//! Locates the eBPF object files that realtime capture loads at runtime.
2//!
3//! A release install keeps the objects beside the executable's prefix in
4//! `<prefix>/lib/rustiflow/`, while development builds leave them in Cargo's
5//! target directory. Both layouts must work without rebuilding the binary.
6
7use std::path::{Path, PathBuf};
8
9use anyhow::bail;
10
11pub(super) const INGRESS_EXPERIMENT_OBJECT: &str = "rustiflow-ebpf-ingress-experiment";
12
13/// Overrides every other location. When set, only this directory is searched,
14/// so a typo fails loudly instead of silently loading a different build.
15pub const EBPF_DIR_ENV: &str = "RUSTIFLOW_EBPF_DIR";
16
17/// Where the release bundle installs the objects, relative to the prefix.
18const INSTALLED_SUBDIR: &str = "lib/rustiflow";
19
20/// eBPF target the userspace loader expects; the only supported one today.
21const EBPF_TARGET: &str = "bpfel-unknown-none";
22
23/// Resolves `program_name` to an existing object file from the current
24/// process environment, or reports every directory that was searched.
25pub fn ebpf_object_path(program_name: &str) -> Result<PathBuf, anyhow::Error> {
26    let search_dirs = ebpf_search_dirs(
27        std::env::var_os(EBPF_DIR_ENV).map(PathBuf::from),
28        std::env::current_exe().ok().as_deref(),
29        &development_target_dir(),
30    );
31
32    if let Some(path) = search_dirs
33        .iter()
34        .map(|dir| dir.join(program_name))
35        .find(|path| path.is_file())
36    {
37        return Ok(path);
38    }
39
40    let searched = search_dirs
41        .iter()
42        .map(|dir| format!("  {}", dir.display()))
43        .collect::<Vec<_>>()
44        .join("\n");
45    if program_name == "rustiflow-ebpf-dual-stack" {
46        bail!(
47            "eBPF object {program_name} not found. Searched:\n{searched}\n\
48             Build the experimental object using docs/profiling/experiments.md \
49             and set {EBPF_DIR_ENV} to its directory."
50        );
51    }
52    let build = if program_name == INGRESS_EXPERIMENT_OBJECT {
53        "cd ebpf-ipv6 && cargo build --locked --release --features xdp-experiment \
54         --bin rustiflow-ebpf-ingress-experiment"
55            .to_owned()
56    } else {
57        format!(
58            "cargo xtask {}",
59            program_name.trim_start_matches("rustiflow-")
60        )
61    };
62    bail!(
63        "eBPF object {program_name} not found. Searched:\n{searched}\n\
64         Set {EBPF_DIR_ENV} to an object directory, or build it with `{build}`."
65    )
66}

68/// Directories to search, in priority order: the explicit override alone, or
69/// else the installed layout followed by the development target directory.
70fn ebpf_search_dirs(
71    override_dir: Option<PathBuf>,
72    executable: Option<&Path>,
73    development_target_dir: &Path,
74) -> Vec<PathBuf> {
75    if let Some(dir) = override_dir {
76        return vec![dir];
77    }
78
79    let installed = executable
80        .and_then(Path::parent)
81        .and_then(Path::parent)
82        .map(|prefix| prefix.join(INSTALLED_SUBDIR));
83    let profile = if cfg!(debug_assertions) {
84        "debug"
85    } else {
86        "release"
87    };
88    let development = development_target_dir.join(EBPF_TARGET).join(profile);
89
90    installed.into_iter().chain([development]).collect()
91}
92
93/// Cargo's target directory for `cargo xtask run` and other source-tree runs.
94fn development_target_dir() -> PathBuf {
95    std::env::var_os("CARGO_TARGET_DIR")
96        .map(PathBuf::from)
97        .unwrap_or_else(|| PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../target"))
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    fn profile() -> &'static str {
105        if cfg!(debug_assertions) {
106            "debug"
107        } else {
108            "release"
109        }
110    }
111
112    #[test]
113    fn override_dir_is_the_only_candidate() {
114        let dirs = ebpf_search_dirs(
115            Some(PathBuf::from("/opt/ebpf")),
116            Some(Path::new("/usr/bin/rustiflow")),
117            Path::new("/src/target"),
118        );
119
120        assert_eq!(dirs, vec![PathBuf::from("/opt/ebpf")]);
121    }
122
123    #[test]
124    fn installed_layout_precedes_development_target() {
125        let dirs = ebpf_search_dirs(
126            None,
127            Some(Path::new("/usr/local/bin/rustiflow")),
128            Path::new("/src/target"),
129        );
130
131        assert_eq!(
132            dirs,
133            vec![
134                PathBuf::from("/usr/local/lib/rustiflow"),
135                PathBuf::from(format!("/src/target/bpfel-unknown-none/{}", profile())),
136            ]
137        );
138    }
139
140    #[test]
141    fn unknown_executable_falls_back_to_development_target() {
142        let dirs = ebpf_search_dirs(None, None, Path::new("/src/target"));
143
144        assert_eq!(
145            dirs,
146            vec![PathBuf::from(format!(
147                "/src/target/bpfel-unknown-none/{}",
148                profile()
149            ))]
150        );
151    }
152
153    #[test]
154    fn missing_objects_report_every_searched_dir() {
155        let error = ebpf_object_path("rustiflow-ebpf-does-not-exist")
156            .expect_err("object must not exist")
157            .to_string();
158
159        assert!(error.contains("rustiflow-ebpf-does-not-exist not found"));
160        assert!(error.contains("cargo xtask ebpf-does-not-exist"));
161    }
162}
