170    /// Decodes an IPv4 packet that starts at `packet`. Bytes past the
171    /// declared total length, such as Ethernet padding, are ignored.
172    pub fn from_ipv4_packet(
173        packet: &[u8],
174        timestamp_us: i64,
175        capture_tcp_payload: bool,
176    ) -> Option<Self> {
177        let (header, _) = Ipv4Header::ref_from_prefix(packet).ok()?;
178        if header.fragment_offset() > 0 {
179            return None;
180        }
181
182        let header_length = header.header_length();
183        let length = header.total_length.get();
184        let transport_length = if capture_tcp_payload && !header.more_fragments() {
185            usize::from(length).checked_sub(header_length)
186        } else {
187            None
188        };
189        // An IHL below the minimum still leaves the fixed header in front.
190        let payload = packet
191            .get(header_length.max(size_of::<Ipv4Header>())..)
192            .map_or(&[][..], |rest| {
193                &rest[..rest
194                    .len()
195                    .min(usize::from(length).saturating_sub(header_length))]
196            });
197        extract_packet_features_transport(
198            Self {
199                source_ip: Ipv4Addr::from(header.source).into(),
200                destination_ip: Ipv4Addr::from(header.destination).into(),
201                protocol: header.protocol,
202                timestamp_us,
203                length,
204                ttl: header.ttl,
205                ..Self::default()
206            },
207            payload,
208            transport_length,
209        )
210    }

212    /// Decodes an IPv6 packet that starts at `packet`. Bytes past the
213    /// declared payload length are ignored.
214    pub fn from_ipv6_packet(
215        packet: &[u8],
216        timestamp_us: i64,
217        capture_tcp_payload: bool,
218    ) -> Option<Self> {
219        let (header, rest) = Ipv6Header::ref_from_prefix(packet).ok()?;
220        let declared_length = header.payload_length.get();
221        let ip_payload = &rest[..rest.len().min(usize::from(declared_length))];
222        let (protocol, payload, fragmented) =
223            skip_ipv6_extension_headers(header.next_header, ip_payload)?;
224        let transport_length = if capture_tcp_payload && !fragmented {
225            usize::from(declared_length).checked_sub(ip_payload.len() - payload.len())
226        } else {
227            None
228        };
229        extract_packet_features_transport(
230            Self {
231                source_ip: Ipv6Addr::from(header.source).into(),
232                destination_ip: Ipv6Addr::from(header.destination).into(),
233                protocol,
234                timestamp_us,
235                length: if capture_tcp_payload {
236                    declared_length.saturating_add(40)
237                } else {
238                    packet.len() as u16
239                },
240                ttl: header.hop_limit,
241                ..Self::default()
242            },
243            payload,
244            transport_length,
245        )
246    }

263fn get_tcp_flag(value: u8, flag: u8) -> u8 {
264    ((value & flag) != 0) as u8

272fn skip_ipv6_extension_headers(
273    mut next_header: u8,
274    mut payload: &[u8],
275) -> Option<(u8, &[u8], bool)> {
276    const MAX_EXTENSION_HEADERS: usize = 8;
277    let mut fragmented = false;
278
279    for _ in 0..MAX_EXTENSION_HEADERS {
280        let Some(extension) = Ipv6Extension::from_next_header(next_header) else {
281            return Some((next_header, payload, fragmented));
282        };
283        let prefix = *payload.first_chunk::<4>()?;
284        let header_len = extension.header_length(prefix)?;
285        fragmented |= matches!(extension, Ipv6Extension::Fragment) && prefix[3] & 1 != 0;
286
287        next_header = prefix[0];
288        payload = payload.get(header_len..)?;
289    }
290
291    Some((next_header, payload, fragmented))
292}

294fn extract_packet_features_transport(
295    mut features: PacketFeatures,
296    packet: &[u8],
297    declared_transport_length: Option<usize>,
298) -> Option<PacketFeatures> {
299    match features.protocol {
300        ip_protocol::TCP => {
301            let (tcp, _) = TcpHeader::ref_from_prefix(packet).ok()?;
302            features.source_port = tcp.source_port.get();
303            features.destination_port = tcp.destination_port.get();
304            features.flags = tcp.flags;
305            features.fin_flag = get_tcp_flag(features.flags, FIN_FLAG);
306            features.syn_flag = get_tcp_flag(features.flags, SYN_FLAG);
307            features.rst_flag = get_tcp_flag(features.flags, RST_FLAG);
308            features.psh_flag = get_tcp_flag(features.flags, PSH_FLAG);
309            features.ack_flag = get_tcp_flag(features.flags, ACK_FLAG);
310            features.urg_flag = get_tcp_flag(features.flags, URG_FLAG);
311            features.cwr_flag = get_tcp_flag(features.flags, CWR_FLAG);
312            features.ece_flag = get_tcp_flag(features.flags, ECE_FLAG);
313            features.header_length = tcp.header_length();
314            // A data offset below the minimum still leaves the fixed header in front.
315            let payload_start = usize::from(features.header_length).max(size_of::<TcpHeader>());
316            features.data_length = packet.len().saturating_sub(payload_start) as u16;
317            features.window_size = tcp.window.get();
318            features.sequence_number = tcp.sequence_number.get();
319            features.sequence_number_ack = tcp.acknowledgement_number.get();
320            if let Some(end) = declared_transport_length {
321                let start = usize::from(features.header_length);
322                if let Some(length) = end.checked_sub(start).filter(|_| start >= 20) {
323                    features.data_length = length as u16;
324                    if length <= 16 * 1024
325                        && crate::flows::features::application::ApplicationProtocol::from_ports(
326                            features.source_port,
327                            features.destination_port,
328                        )
329                        .is_some()
330                    {
331                        features.tcp_payload = packet.get(start..end).map(Box::from);
332                    }
333                }
334            }
335        }
336        ip_protocol::UDP => {
337            let (udp, payload) = UdpHeader::ref_from_prefix(packet).ok()?;
338            features.source_port = udp.source_port.get();
339            features.destination_port = udp.destination_port.get();
340            features.data_length = payload.len() as u16;
341            features.header_length = 8;
342        }
343        ip_protocol::ICMP | ip_protocol::ICMPV6 => {
344            // Like the realtime programs, require and count the full 8-byte
345            // header, including its message-specific second word.
346            let (icmp, payload) = IcmpMessageHeader::ref_from_prefix(packet).ok()?;
347            features.data_length = payload.len() as u16;
348            features.header_length = size_of::<IcmpMessageHeader>() as u8;
349            features.icmp_type = Some(icmp.header.icmp_type);
350            features.icmp_code = Some(icmp.header.code);
351        }
352        _ => {
353            debug!("Unsupported protocol in packet!");
354            return None;
355        }
356    }
357    Some(features)
