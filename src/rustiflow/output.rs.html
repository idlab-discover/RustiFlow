20/// Flows a shard may queue for its encoder before the shard waits.
21const SHARD_FLOW_QUEUE_CAPACITY: usize = 256;
22/// Flows an encoder takes from its queue per wakeup.
23const ENCODER_FLOW_BATCH: usize = 64;
24/// Encoded bytes an encoder accumulates before handing a chunk to the writer.
25const ENCODED_CHUNK_BYTES: usize = 64 * 1024;
26/// Encoded chunks that may wait for the writer.
27const ENCODED_CHUNK_QUEUE_CAPACITY: usize = 64;

29pub enum FlowExporter<T> {
30    Parquet(parquet::Exporter<T>),
31    Csv {
32        clean: bool,
33        chunks: mpsc::Sender<String>,
34    },
35}

37/// One flow table's output. Parquet projects directly from the owned flow
38/// while it is still hot in the table worker's cache.
39pub enum ShardOutput<T> {
40    Channel(mpsc::Sender<T>),
41    Parquet(parquet::Shard<T>),
42}
43
44impl<T> From<mpsc::Sender<T>> for ShardOutput<T> {
45    fn from(sender: mpsc::Sender<T>) -> Self {
46        Self::Channel(sender)
47    }
48}

50impl<T: Flow> ShardOutput<T> {
51    pub async fn export(&mut self, flow: T) {
52        match self {
53            Self::Channel(sender) => {
54                if sender.send(flow).await.is_err() {
55                    error!("Failed to send flow: export channel is closed");
56                }
57            }
58            Self::Parquet(shard) => shard.export(&flow).await,
59        }
60    }
61
62    pub async fn flush(&mut self) {
63        if let Self::Parquet(shard) = self {
64            shard.flush().await;
65        }
66    }
67}

69/// Destination for the flows that each shard's flow table exports.
70pub trait FlowSink<T>: Send + 'static {
71    /// Creates the output used by one shard for its lifetime.
72    fn shard_output(&self) -> ShardOutput<T>;
73}

75/// All shards share the channel; used by tests that inspect exported flows.
76impl<T: Flow> FlowSink<T> for mpsc::Sender<T> {
77    fn shard_output(&self) -> ShardOutput<T> {
78        self.clone().into()
79    }

82impl<T: Flow> FlowSink<T> for FlowExporter<T> {
83    fn shard_output(&self) -> ShardOutput<T> {
84        let (clean, chunk_sender) = match self {
85            Self::Parquet(exporter) => return ShardOutput::Parquet(exporter.shard()),
86            Self::Csv { clean, chunks } => (*clean, chunks.clone()),
87        };
88        let (flow_sender, mut flow_receiver) = mpsc::channel::<T>(SHARD_FLOW_QUEUE_CAPACITY);
89
90        tokio::spawn(async move {
91            let mut flows = Vec::with_capacity(ENCODER_FLOW_BATCH);
92            let mut chunk = String::with_capacity(ENCODED_CHUNK_BYTES);
93
94            while flow_receiver
95                .recv_many(&mut flows, ENCODER_FLOW_BATCH)
96                .await
97                > 0
98            {
99                for flow in flows.drain(..) {
100                    let dump_start = Instant::now();
101                    let start = chunk.len();
102                    project(&flow, clean, &mut CsvRow::new(&mut chunk));
103                    export_profile::record_dump(dump_start.elapsed(), chunk.len() - start);
104                    chunk.push('\n');
105                }
106
107                // Hand over full chunks, and partial ones when idle, so rows of a
108                // quiet shard are not held back.
109                if chunk.len() >= ENCODED_CHUNK_BYTES || flow_receiver.is_empty() {
110                    let full_chunk =
111                        std::mem::replace(&mut chunk, String::with_capacity(ENCODED_CHUNK_BYTES));
112                    if chunk_sender.send(full_chunk).await.is_err() {
113                        error!("Failed to send encoded flows: output writer is closed");
114                        return;
115                    }
116                }
117            }
118
119            if !chunk.is_empty() && chunk_sender.send(chunk).await.is_err() {
120                error!("Failed to send encoded flows: output writer is closed");
121            }
122        });
123
124        flow_sender.into()
125    }
126}

128fn project<T: Flow>(flow: &T, clean: bool, row: &mut impl Record) {
129    if clean {
130        flow.write_clean_fields(row);
131    } else {
132        flow.write_fields(row);
133    }
134}

136pub fn spawn_output_writer<T: Flow>(
137    output: OutputConfig,
138) -> Result<(FlowExporter<T>, JoinHandle<Result<()>>)> {
139    let clean = output.drop_contaminant_features;
140    let headers = if clean {
141        T::get_features_without_contamination()
142    } else {
143        T::get_features()
144    };
145    let open =
146        |path: &str| File::create(path).with_context(|| format!("creating output file {path}"));
147    let destination: Box<dyn Write + Send> = match output.output {
148        ExportMethodType::Parquet => {
149            let file = match output.export_path.as_deref() {
150                Some(path) => open(path)?,
151                None => File::create_new("output.parquet").context(
152                    "creating output.parquet (set --export-path to choose a destination)",
153                )?,
154            };
155            return parquet::spawn::<T>(file, &headers, clean);
156        }
157        ExportMethodType::Csv => Box::new(open(
158            output
159                .export_path
160                .as_deref()
161                .context("CSV output requires an export path")?,
162        )?),
163        ExportMethodType::Print => Box::new(std::io::stdout()),
164    };
165    let mut writer = BufWriter::new(destination);
166    if output.header {
167        writeln!(writer, "{headers}")?;
168    }
169    let (chunks, mut receiver) = mpsc::channel::<String>(ENCODED_CHUNK_QUEUE_CAPACITY);
170    let task = tokio::task::spawn_blocking(move || {
171        while let Some(rows) = receiver.blocking_recv() {
172            let start = Instant::now();
173            writer.write_all(rows.as_bytes())?;
174            export_profile::record_write(start.elapsed());
175        }
176        writer.flush()?;
177        Ok(())
178    });
179    Ok((FlowExporter::Csv { clean, chunks }, task))
180}
