15#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq)]
16pub struct FlowKey {
17    pub source: EndpointKey,
18    pub destination: EndpointKey,
19    pub protocol: u8,
20}
21
22impl FlowKey {
23    pub fn new(
24        source_ip: IpAddr,
25        source_port: u16,
26        destination_ip: IpAddr,
27        destination_port: u16,
28        protocol: u8,
29    ) -> Self {
30        Self {
31            source: EndpointKey::new(source_ip, source_port),
32            destination: EndpointKey::new(destination_ip, destination_port),
33            protocol,
34        }
35    }
36
37    pub fn reverse(self) -> Self {
38        Self {
39            source: self.destination,
40            destination: self.source,
41            protocol: self.protocol,
42        }
43    }
44
45    pub fn canonical(self) -> Self {
46        self.canonicalize().0
47    }
48
49    /// Returns the canonical biflow key together with the orientation `self`
50    /// had relative to it. A key whose endpoints are equal is canonical.
51    pub fn canonicalize(self) -> (Self, KeyOrientation) {
52        if self.source <= self.destination {
53            (self, KeyOrientation::Canonical)
54        } else {
55            (self.reverse(), KeyOrientation::Reversed)
56        }
57    }
58}

60/// Direction of a packet's 5-tuple relative to its canonical biflow key.
61#[derive(Clone, Copy, Debug, Eq, PartialEq)]
62pub enum KeyOrientation {
63    Canonical,
64    Reversed,
65}

81/// Hash builder for flow-keyed state.
82///
83/// Every instance draws its own seed on top of a random per-process seed, so
84/// remote traffic cannot precompute colliding keys, and independently built
85/// instances (for example a shard selector and a shard's flow map) do not
86/// correlate their bucket choices.
87pub type FlowKeyHashBuilder = foldhash::quality::RandomState;
88
89/// Assigns biflow keys to a fixed set of shards.
90#[derive(Clone)]
91pub struct ShardSelector {
92    hash_builder: FlowKeyHashBuilder,
93    shard_count: NonZeroUsize,
94}
95
96impl ShardSelector {
97    pub fn new(shard_count: NonZeroUsize) -> Self {
98        Self {
99            hash_builder: FlowKeyHashBuilder::default(),
100            shard_count,
101        }
102    }
103
104    /// Returns the shard for `biflow_key`, in `0..shard_count`. Callers must pass
105    /// the canonical key so both directions of a flow land on the same shard.
106    pub fn shard_index(&self, biflow_key: &FlowKey) -> usize {
107        (self.hash_builder.hash_one(biflow_key) % self.shard_count.get() as u64) as usize
108    }
