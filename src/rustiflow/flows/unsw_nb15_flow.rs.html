29#[derive(Clone, Default)]
30struct Direction {
31    bytes: i64,
32    packets: i64,
33    first_ttl: u8,
34    first_window: u16,
35    first_sequence: u32,
36}
37
38impl Direction {
39    fn observe(&mut self, packet: &PacketFeatures) {
40        if self.packets == 0 {
41            self.first_ttl = packet.ttl;
42            self.first_window = packet.window_size;
43            self.first_sequence = packet.sequence_number;
44        }
45        self.bytes += i64::from(packet.length);
46        self.packets += 1;
47    }
48
49    fn mean_size(&self) -> i64 {
50        self.bytes.checked_div(self.packets).unwrap_or(0)
51    }
52
53    fn load(&self, duration: f64) -> f64 {
54        safe_div((self.bytes - self.mean_size()) as f64 * 8.0, duration)
55    }
56}

73impl UnswNb15Flow {
74    fn protocol_name(&self) -> &'static str {
75        match self.basic.protocol {
76            1 => "icmp",
77            6 => "tcp",
78            17 => "udp",
79            58 => "ipv6-icmp",
80            _ => "-",
81        }
82    }
83
84    fn state(&self) -> &'static str {
85        let bidirectional = self.source.packets > 0 && self.destination.packets > 0;
86        match self.basic.protocol {
87            6 => match self.basic.tcp_close_style {
88                TcpCloseStyle::Reset => "RST",
89                TcpCloseStyle::FourWayFin | TcpCloseStyle::SimultaneousFin => "FIN",
90                TcpCloseStyle::BidirectionalFin | TcpCloseStyle::HalfClose => "CLO",
91                TcpCloseStyle::None | TcpCloseStyle::NotApplicable => {
92                    if self.basic.tcp_handshake_completed {
93                        "CON"
94                    } else if matches!(self.setup, TcpSetup::SynAck { .. }) {
95                        "ACC"
96                    } else if matches!(self.setup, TcpSetup::Syn { .. }) {
97                        "REQ"
98                    } else if bidirectional {
99                        "CON"
100                    } else {
101                        "INT"
102                    }
103                }
104            },
105            1 | 58 => match (self.basic.protocol, self.first_icmp_type) {
106                (1, Some(8)) | (58, Some(128)) => "ECO",
107                (1, Some(0)) | (58, Some(129)) => "ECR",
108                (1, Some(3)) | (58, Some(1)) => "URN",
109                (1, Some(11)) | (58, Some(3)) => "TXD",
110                _ => "-",
111            },
112            17 if bidirectional => "CON",
113            17 => "INT",
114            _ => "-",
115        }
116    }
117
118    fn write(&self, row: &mut impl Record, clean: bool) {
119        let basic = &self.basic;
120        let src = &self.source;
121        let dst = &self.destination;
122        let duration = basic.get_flow_duration_usec() as f64 / 1_000_000.0;
123        let state = self.state();
124        if !clean {
125            fields!(
126                row,
127                basic.ip_source,
128                basic.port_source,
129                basic.ip_destination,
130                basic.port_destination
131            );
132        }
133        fields!(
134            row,
135            self.protocol_name(),
136            state,
137            duration,
138            src.bytes,
139            dst.bytes,
140            src.first_ttl,
141            dst.first_ttl,
142            self.retransmissions.fwd_retransmission_count,
143            self.retransmissions.bwd_retransmission_count,
144            self.service,
145            src.load(duration),
146            dst.load(duration),
147            src.packets,
148            dst.packets,
149            src.first_window,
150            dst.first_window,
151        );
152        if !clean {
153            fields!(row, src.first_sequence, dst.first_sequence);
154        }
155        let app = self
156            .application
157            .as_deref()
158            .map_or_else(Default::default, ApplicationStats::fields);
159        fields!(
160            row,
161            src.mean_size(),
162            dst.mean_size(),
163            app.transaction_depth,
164            app.response_body_length,
165            self.iat.fwd_iat.get_std(),
166            self.iat.bwd_iat.get_std(),
167        );
168        if !clean {
169            fields!(
170                row,
171                basic.first_timestamp_us.div_euclid(1_000_000),
172                basic.last_timestamp_us.div_euclid(1_000_000),
173            );
174        }
175        let (synack, ackdat) = self.setup.intervals_seconds();
176        let same_endpoint =
177            basic.ip_source == basic.ip_destination && basic.port_source == basic.port_destination;
178        fields!(
179            row,
180            self.iat.fwd_iat.get_mean(),
181            self.iat.bwd_iat.get_mean(),
182            synack + ackdat,
183            synack,
184            ackdat,
185            u8::from(same_endpoint),
186            state_ttl(state, src.first_ttl, dst.first_ttl),
187            app.http_methods,
188            app.ftp_login,
189            app.ftp_commands,
190        );
191        let counts = &self.counts;
192        fields!(
193            row,
194            counts.service_source,
195            counts.service_destination,
196            counts.destination,
197            counts.source,
198            counts.source_destination_port,
199            counts.destination_source_port,
200            counts.source_destination,
201        );
202    }
203}

205fn service_for_port(protocol: u8, port: u16) -> &'static str {
206    match (protocol, port) {
207        (6, 20) => "ftp-data",
208        (6, 21) => "ftp",
209        (6, 22) => "ssh",
210        (6, 25 | 587) => "smtp",
211        (6 | 17, 53) => "dns",
212        (6, 80 | 8000 | 8080) => "http",
213        (6, 110) => "pop3",
214        (6, 143) => "imap",
215        (17, 161 | 162) => "snmp",
216        (6, 443) => "ssl",
217        (17, 67 | 68) => "dhcp",
218        (6, 6667) => "irc",
219        (6, 1812 | 1813) | (17, 1812 | 1813) => "radius",
220        _ => "-",
221    }
222}

224fn state_ttl(state: &str, source: u8, destination: u8) -> u8 {
225    match (state, source, destination) {
226        ("FIN", 62 | 63 | 254 | 255, 252 | 253) => 1,
227        ("INT", 0 | 62 | 254, 0) => 2,
228        ("CON", 62 | 254, 60 | 252 | 253) => 3,
229        ("ACC", 254, 252) => 4,
230        ("CLO", 254, 252) => 5,
231        ("REQ", 254, 0) => 6,
232        _ => 0,
233    }
234}

236impl Flow for UnswNb15Flow {
237    const CAPTURE_TCP_PAYLOAD: bool = true;
238    const USES_CONNECTION_HISTORY: bool = true;

240    fn new(
241        key: String,
242        source: IpAddr,
243        source_port: u16,
244        destination: IpAddr,
245        destination_port: u16,
246        protocol: u8,
247        timestamp_us: i64,
248    ) -> Self {
249        let destination_service = service_for_port(protocol, destination_port);
250        let forward_from_client = destination_service != "-";
251        let service = if forward_from_client {
252            destination_service
253        } else {
254            service_for_port(protocol, source_port)
255        };
256        let application = match service {
257            "http" => Some(ApplicationProtocol::Http),
258            "ftp" => Some(ApplicationProtocol::Ftp),
259            _ => None,
260        }
261        .map(|protocol| Box::new(ApplicationStats::new(protocol)));
262        Self {
263            basic: BasicFlow::new(
264                key,
265                source,
266                source_port,
267                destination,
268                destination_port,
269                protocol,
270                timestamp_us,
271            ),
272            source: Direction::default(),
273            destination: Direction::default(),
274            iat: IATStats::new(),
275            retransmissions: RetransmissionStats::new(),
276            setup: TcpSetup::default(),
277            application,
278            service,
279            forward_from_client,
280            first_icmp_type: None,
281            counts: ConnectionCounts::default(),
282        }
283    }
284
285    fn connection(&self) -> Option<Connection> {
286        let basic = &self.basic;
287        Some(Connection {
288            key: FlowKey::new(
289                basic.ip_source,
290                basic.port_source,
291                basic.ip_destination,
292                basic.port_destination,
293                basic.protocol,
294            ),
295            service: self.service,
296            first_timestamp_us: basic.first_timestamp_us,
297            last_timestamp_us: basic.last_timestamp_us,
298            completed: basic.flow_expire_cause != FlowExpireCause::None,
299        })
300    }
301
302    fn set_connection_counts(&mut self, counts: ConnectionCounts) {
303        self.counts = counts;
304    }
305
306    fn update_flow(&mut self, packet: &PacketFeatures, forward: bool) -> bool {
307        if forward {
308            self.source.observe(packet);
309        } else {
310            self.destination.observe(packet);
311        }
312        self.first_icmp_type = self.first_icmp_type.or(packet.icmp_type);
313        let previous_timestamp = self.basic.last_timestamp_us;
314        self.iat.update(packet, forward, previous_timestamp);
315        self.retransmissions
316            .update(packet, forward, previous_timestamp);
317        if packet.protocol == 6 {
318            self.setup.observe(packet, forward);
319        }
320        if let Some(application) = self.application.as_deref_mut() {
321            application.observe(packet, forward == self.forward_from_client);
322        }
323        self.basic.update_flow(packet, forward)
324    }
325
326    fn close_flow(&mut self, timestamp_us: i64, cause: FlowExpireCause) {
327        self.basic.close_flow(timestamp_us, cause);
328    }
