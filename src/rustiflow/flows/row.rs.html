50/// Append directly to the encoder's chunk, tracking separators per record.
51pub struct CsvRow<'a> {
52    chunk: &'a mut String,
53    first: bool,
54}
55impl<'a> CsvRow<'a> {
56    pub fn new(chunk: &'a mut String) -> Self {
57        Self { chunk, first: true }
58    }
59}
60impl Record for CsvRow<'_> {
61    fn append(&mut self, value: Value<'_>) {
62        if !self.first {
63            self.chunk.push(',');
64        }
65        self.first = false;
66        match value {
67            Value::Integer(n) => push_i64_digits(self.chunk, n),
68            Value::Float(n) => append_float(self.chunk, n),
69            Value::Text(s) => self.chunk.push_str(s),
70            Value::Address(ip) => write!(self.chunk, "{ip}").expect("writing to String"),
71            Value::Timestamp(ts) => write!(self.chunk, "{ts}").expect("writing to String"),
72        }
73    }
74}
75/// Largest magnitude below which every integral `f64` converts to `i64` exactly.
76const EXACT_INTEGER_F64_LIMIT: f64 = 9_007_199_254_740_992.0; // 2^53
77
78/// Appends `value` as a CSV field, byte-identical to its `Display` output.
79///
80/// `Display` renders integral floats without a fractional part (`3`, `-0`), so
81/// integral values take the much cheaper integer formatter. Most exported
82/// statistics (lengths, counts, totals, empty stats) are integral. Negative zero
83/// keeps the float path because `-0` has no `i64` representation.
84fn append_float(output: &mut String, value: f64) {
85    // The round trip through i64 tests integrality without a libm `trunc` call;
86    // NaN fails the range check.
87    let integer = value as i64;
88    if value.abs() < EXACT_INTEGER_F64_LIMIT
89        && integer as f64 == value
90        && !(value == 0.0 && value.is_sign_negative())
91    {
92        push_i64_digits(output, integer);
93    } else {
94        let _ = write!(output, "{value}");
95    }
96}
97
98/// Appends the decimal form of `value`, matching its `Display` output without
99/// going through the `fmt` machinery.
100fn push_i64_digits(output: &mut String, value: i64) {
101    if value < 0 {
102        output.push('-');
103    }
104    let mut remaining = value.unsigned_abs();
105    let mut digits = [0u8; 20]; // u64::MAX has 20 decimal digits
106    let mut start = digits.len();
107    loop {
108        start -= 1;
109        digits[start] = b'0' + (remaining % 10) as u8;
110        remaining /= 10;
111        if remaining == 0 {
112            break;
113        }
114    }
115    output.extend(digits[start..].iter().map(|&digit| char::from(digit)));
116}
