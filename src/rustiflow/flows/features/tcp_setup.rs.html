4pub(crate) enum TcpSetup {
5    #[default]
6    Unseen,
7    Syn {
8        timestamp_us: i64,
9        expected_ack: u32,
10    },
11    SynAck {
12        timestamp_us: i64,
13        synack_us: i64,
14        expected_ack: u32,
15    },
16    Complete {
17        synack_us: i64,
18        ackdat_us: i64,
19    },
20}
21
22impl TcpSetup {
23    /// Measure a sequence-validated three-way handshake using the first
24    /// observed SYN and SYN-ACK. Call only for TCP packets.
25    pub(crate) fn observe(&mut self, packet: &PacketFeatures, forward: bool) {
26        let timestamp_us = packet.timestamp_us;
27        let expected_ack = || {
28            packet
29                .sequence_number
30                .wrapping_add(1 + u32::from(packet.data_length))
31        };
32
33        match *self {
34            Self::Unseen if forward && packet.syn_flag != 0 && packet.ack_flag == 0 => {
35                *self = Self::Syn {
36                    timestamp_us,
37                    expected_ack: expected_ack(),
38                };
39            }
40            Self::Syn {
41                timestamp_us: syn,
42                expected_ack: ack,
43            } if !forward
44                && packet.syn_flag != 0
45                && packet.ack_flag != 0
46                && packet.sequence_number_ack == ack
47                && timestamp_us >= syn =>
48            {
49                *self = Self::SynAck {
50                    timestamp_us,
51                    synack_us: timestamp_us.saturating_sub(syn),
52                    expected_ack: expected_ack(),
53                };
54            }
55            Self::SynAck {
56                timestamp_us: synack,
57                synack_us,
58                expected_ack,
59            } if forward
60                && packet.syn_flag == 0
61                && packet.ack_flag != 0
62                && packet.rst_flag == 0
63                && packet.sequence_number_ack == expected_ack
64                && timestamp_us >= synack =>
65            {
66                *self = Self::Complete {
67                    synack_us,
68                    ackdat_us: timestamp_us.saturating_sub(synack),
69                };
70            }
71            Self::Unseen | Self::Syn { .. } | Self::SynAck { .. } | Self::Complete { .. } => {}
72        }
73    }
74
75    pub(crate) fn intervals_seconds(&self) -> (f64, f64) {
76        let (synack, ackdat) = match *self {
77            Self::Unseen | Self::Syn { .. } => (0, 0),
78            Self::SynAck { synack_us, .. } => (synack_us, 0),
79            Self::Complete {
80                synack_us,
81                ackdat_us,
82            } => (synack_us, ackdat_us),
83        };
84        (synack as f64 / 1_000_000.0, ackdat as f64 / 1_000_000.0)
85    }
