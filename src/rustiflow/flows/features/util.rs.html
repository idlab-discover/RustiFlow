5/// Trait for network flow features that can be updated, closed, and dumped to CSV format
6pub trait FlowFeature: Send + Sync + Clone {
7    /// Updates the feature with a new packet
8    fn update(&mut self, packet: &PacketFeatures, is_forward: bool, last_timestamp: i64);
9
10    /// Finalizes any active state when the flow is terminated
11    fn close(&mut self, last_timestamp: i64, cause: FlowExpireCause);
12
13    /// Appends the current state as CSV fields to an existing row buffer.
14    fn append_to_csv(&self, output: &mut String);
15
16    /// Dumps the current state as a standalone CSV string.
17    #[cfg(test)]
18    fn dump(&self) -> String {
19        let mut output = String::new();
20        self.append_to_csv(&mut output);
21        output
22    }
23
24    /// Returns the CSV headers for this feature
25    fn headers() -> String
26    where
27        Self: Sized;
28}

88    fn update_min(&mut self, value: f64) {
89        if value < self.min {
90            self.min = value;
91        }
92    }

94    fn update_max(&mut self, value: f64) {
95        if value > self.max {
96            self.max = value;
97        }
98    }

100    pub fn add_value(&mut self, value: f64) {
101        self.total += value;
102        self.update_min(value);
103        self.update_max(value);
104        self.count += 1;
105
106        let delta = value - self.mean;
107        self.mean += delta / self.count as f64;
108        let delta2 = value - self.mean;
109        self.m2 += delta * delta2;
110    }

122    }

132/// Largest magnitude below which every integral `f64` converts to `i64` exactly.
133const EXACT_INTEGER_F64_LIMIT: f64 = 9_007_199_254_740_992.0; // 2^53
134
135/// Appends `value` as a CSV field, byte-identical to its `Display` output.
136///
137/// `Display` renders integral floats without a fractional part (`3`, `-0`), so
138/// integral values take the much cheaper integer formatter. Most exported
139/// statistics (lengths, counts, totals, empty stats) are integral. Negative zero
140/// keeps the float path because `-0` has no `i64` representation.
141pub fn push_csv_f64(output: &mut String, value: f64) {
142    if !output.is_empty() {
143        output.push(',');
144    }
145    // The round trip through i64 tests integrality without a libm `trunc` call;
146    // NaN fails the range check.
147    let integer = value as i64;
148    if value.abs() < EXACT_INTEGER_F64_LIMIT
149        && integer as f64 == value
150        && !(value == 0.0 && value.is_sign_negative())
151    {
152        push_i64_digits(output, integer);
153    } else {
154        let _ = write!(output, "{value}");
155    }
156}

158/// Appends an integer CSV field, byte-identical to its `Display` output.
159pub fn push_csv_int(output: &mut String, value: impl Into<i64>) {
160    if !output.is_empty() {
161        output.push(',');
162    }
163    push_i64_digits(output, value.into());
164}
165
166/// Appends the decimal form of `value`, matching its `Display` output without
167/// going through the `fmt` machinery.
168fn push_i64_digits(output: &mut String, value: i64) {
169    if value < 0 {
170        output.push('-');
171    }
172    let mut remaining = value.unsigned_abs();
173    let mut digits = [0u8; 20]; // u64::MAX has 20 decimal digits
174    let mut start = digits.len();
175    loop {
176        start -= 1;
177        digits[start] = b'0' + (remaining % 10) as u8;
178        remaining /= 10;
179        if remaining == 0 {
180            break;
181        }
182    }
183    output.extend(digits[start..].iter().map(|&digit| char::from(digit)));
184}

186/// Safely performs floating point division, returning 0.0 if denominator is 0
187pub fn safe_div(numerator: f64, denominator: f64) -> f64 {
188    if denominator == 0.0 {
189        0.0
190    } else {
191        numerator / denominator
192    }
193}
194
195/// Safely performs integer division, returning 0.0 if denominator is 0
196pub fn safe_div_int(numerator: u32, denominator: u32) -> f64 {
197    if denominator == 0 {
198        0.0
199    } else {
200        numerator as f64 / denominator as f64
201    }
202}

204/// Safely calculates per-second rate, handling zero duration
205pub fn safe_per_second_rate(value: f64, duration_usec: f64) -> f64 {
206    safe_div(value, duration_usec / 1_000_000.0)
