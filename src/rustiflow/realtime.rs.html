92fn log_source_stats(label: &str, stats: &RealtimeSourceStats) {
93    let events = stats.events.load(Ordering::Relaxed);
94    let decode_and_shard_ns = stats.decode_and_shard_ns.load(Ordering::Relaxed);
95    let dispatch_enqueue_ns = stats.dispatch_enqueue_ns.load(Ordering::Relaxed);
96    let shard_send_wait_ns = stats.shard_send_wait_ns.load(Ordering::Relaxed);
97    let total_event_ns = stats.total_event_ns.load(Ordering::Relaxed);
98    let send_errors = stats.send_errors.load(Ordering::Relaxed);
99
100    if events == 0 {
101        info!("Realtime source {}: no events drained", label);
102        return;
103    }
104
105    info!(
106        "Realtime source {}: events={} total_ms={:.3} decode_ms={:.3} enqueue_wait_ms={:.3} shard_send_wait_ms={:.3} avg_event_us={:.3} avg_enqueue_wait_us={:.3} avg_shard_send_wait_us={:.3} send_errors={}",
107        label,
108        events,
109        total_event_ns as f64 / 1_000_000.0,
110        decode_and_shard_ns as f64 / 1_000_000.0,
111        dispatch_enqueue_ns as f64 / 1_000_000.0,
112        shard_send_wait_ns as f64 / 1_000_000.0,
113        total_event_ns as f64 / events as f64 / 1_000.0,
114        dispatch_enqueue_ns as f64 / events as f64 / 1_000.0,
115        shard_send_wait_ns as f64 / events as f64 / 1_000.0,
116        send_errors,
117    );
118}

157/// Spawns one flow-table worker per shard. Each worker exports its remaining
158/// flows once every sender for its shard is dropped.
159fn spawn_shard_workers<T: Flow>(
160    shard_count: NonZeroUsize,
161    shard_queue_capacity: usize,
162    flow_table_settings: &FlowTableSettings,
163    exporter: &impl FlowSink<T>,
164) -> Vec<Sender<Vec<PacketFeatures>>> {
165    debug!("Creating {} sharded FlowTables...", shard_count);
166    let shard_senders = (0..shard_count.get())
167        .map(|_| {
168            let (tx, mut rx) = mpsc::channel::<Vec<PacketFeatures>>(shard_queue_capacity);
169            let mut flow_table = FlowTable::new(
170                flow_table_settings.active_timeout,
171                flow_table_settings.idle_timeout,
172                flow_table_settings.early_export,
173                exporter.shard_sender(),
174                flow_table_settings.expiration_check_interval,
175            );
176
177            tokio::spawn(async move {
178                let mut last_timestamp = None;
179                while let Some(packet_batch) = rx.recv().await {
180                    for packet_features in packet_batch {
181                        last_timestamp = Some(packet_features.timestamp_us);
182                        flow_table.process_packet(&packet_features).await;
183                    }
184                }
185                debug!("Shard finished processing packets");
186                // Handle flow exporting when the receiver is closed
187                if let Some(timestamp) = last_timestamp {
188                    flow_table.export_all_flows(timestamp).await;
189                }
190            });
191            tx
192        })
193        .collect();
194    debug!("Sharded FlowTables created");
195    shard_senders
196}

201fn decode_ipv4_event(event: &[u8], realtime_offset_us: i64) -> PacketFeatures {
202    // SAFETY: the IPv4 programs only submit `EbpfEventIpv4` records to their rings.

204    PacketFeatures::from_ebpf_event_ipv4(&event, realtime_offset_us)
205}

213/// One event source's userspace stage: pick a shard per packet, batch per
214/// shard, and hand batches to the source's own dispatcher task so draining
215/// never waits on a shard queue directly.
216struct SourceBatcher {
217    shard_selector: ShardSelector,
218    shard_batch_size: usize,
219    pending_batches: Vec<Vec<PacketFeatures>>,
220    dispatch_sender: Sender<ShardDispatchBatch>,
221    stats: Option<Arc<RealtimeSourceStats>>,
222}
223
224impl SourceBatcher {
225    /// Creates the batcher and spawns its dispatcher into `handle_set`.
226    fn new(
227        handle_set: &mut JoinSet<()>,
228        shard_senders: &[Sender<Vec<PacketFeatures>>],
229        shard_selector: &ShardSelector,
230        settings: &PipelineSettings,
231        stats: Option<Arc<RealtimeSourceStats>>,
232    ) -> Self {
233        let dispatch_sender = spawn_source_dispatcher(
234            handle_set,
235            shard_senders.to_vec(),
236            settings.source_dispatch_queue_capacity,
237            stats.clone(),
238        );
239        Self {
240            shard_selector: shard_selector.clone(),
241            shard_batch_size: settings.shard_batch_size,
242            pending_batches: create_pending_batches(shard_senders.len(), settings.shard_batch_size),
243            dispatch_sender,
244            stats,
245        }
246    }
247
248    /// Decodes one event and queues it for its shard, dispatching the shard's
249    /// batch once it is full.
250    async fn push_event(&mut self, decode: impl FnOnce() -> PacketFeatures) {
251        let event_start = self.stats.is_some().then(Instant::now);
252        let packet_features = decode();
253        let shard_index = self
254            .shard_selector
255            .shard_index(&packet_features.biflow_key_value());
256        if let (Some(stats), Some(event_start)) = (&self.stats, event_start) {
257            stats.add_decode_and_shard_ns(elapsed_ns(event_start));
258        }
259
260        let pending_batch = &mut self.pending_batches[shard_index];
261        pending_batch.push(packet_features);
262        if pending_batch.len() >= self.shard_batch_size {
263            enqueue_shard_batch(
264                &self.dispatch_sender,
265                pending_batch,
266                self.stats.as_ref(),
267                shard_index,
268            )
269            .await;
270        }
271
272        if let (Some(stats), Some(event_start)) = (&self.stats, event_start) {
273            stats.increment_events();
274            stats.add_total_event_ns(elapsed_ns(event_start));
275        }
276    }
277
278    /// Dispatches every partial batch, as a source does when its ring runs empty.
279    async fn flush(&mut self) {
280        enqueue_pending_batches(
281            &self.dispatch_sender,
282            &mut self.pending_batches,
283            self.stats.as_ref(),
284        )
285        .await;
286    }
287}

289/// Drains one ring buffer for as long as the task runs.
290async fn drain_ring_source(
291    ring_buf: RingBuf<MapData>,
292    mut batcher: SourceBatcher,
293    packet_graph: Option<PacketGraphState>,
294    decode_event: DecodeEvent,
295    realtime_offset_us: i64,
296) {
297    // Wrap the RingBuf in AsyncFd to poll it with tokio.
298    // SAFETY: the ring buffer owns its map file descriptor and reports that
299    // same descriptor from `as_raw_fd` until it is dropped with the `AsyncFd`.
300    let mut async_ring_buf = unsafe { AsyncFd::register(ring_buf) }.unwrap();
301
302    loop {
303        // Wait for data to be available in the ring buffer
304        let mut guard = async_ring_buf.readable_mut().await.unwrap();
305
306        let ring_buf = guard.get_inner_mut();
307        let mut drained_events = 0;
308        while let Some(event) = ring_buf.next() {
309            drained_events += 1;
310            if drained_events == PACKET_GRAPH_RECORD_BATCH {
311                if let Some(packet_graph) = &packet_graph {
312                    packet_graph.record_packets(drained_events);
313                }
314                drained_events = 0;
315            }
316            batcher
317                .push_event(|| decode_event(&event, realtime_offset_us))
318                .await;
319        }
320
321        if let Some(packet_graph) = &packet_graph {
322            packet_graph.record_packets(drained_events);
323        }
324        batcher.flush().await;
325
326        // Clear the readiness state for the next iteration
327        guard.clear_ready();
328    }
329}

331/// Starts the realtime processing of packets on the given interface.
332/// The function will return the number of packets dropped by the eBPF program.
333#[allow(clippy::too_many_arguments)]
334pub async fn handle_realtime<T>(
335    interface: &str,
336    exporter: impl FlowSink<T>,
337    num_threads: u8,
338    active_timeout: u64,
339    idle_timeout: u64,
340    early_export: Option<NonZeroU64>,
341    expiration_check_interval: u64,
342    ingress_only: bool,
343    packet_graph_mode: PacketGraphMode,
344) -> Result<u64, anyhow::Error>
345where
346    T: Flow,
347{
348    let settings = PipelineSettings::from_env();
349    let shard_count =
350        NonZeroUsize::new(usize::from(num_threads)).context("thread count must be at least 1")?;
351    let shard_selector = ShardSelector::new(shard_count);
352
353    // Needed for older kernels
354    bump_memlock_rlimit();
355
356    let realtime_offset_us = compute_realtime_offset_us()?;
357
358    // Load the eBPF programs and attach to the event arrays
359    let mut bpf_ingress_ipv4 = load_ebpf_ipv4(interface, TcAttachType::Ingress)?;
360    let mut bpf_ingress_ipv6 = load_ebpf_ipv6(interface, TcAttachType::Ingress)?;
361    let events_ingress_ipv4 = take_ring_buf_maps(&mut bpf_ingress_ipv4, "EVENTS_IPV4")?;
362    let dropped_packets_ingress_ipv4: PerCpuArray<_, u64> =
363        PerCpuArray::try_from(bpf_ingress_ipv4.take_map("DROPPED_PACKETS").unwrap())?;
364    let matched_packets_ingress_ipv4: PerCpuArray<_, u64> =
365        PerCpuArray::try_from(bpf_ingress_ipv4.take_map("MATCHED_PACKETS").unwrap())?;
366    let submitted_events_ingress_ipv4: PerCpuArray<_, u64> =
367        PerCpuArray::try_from(bpf_ingress_ipv4.take_map("SUBMITTED_EVENTS").unwrap())?;
368    let events_ingress_ipv6 = take_ring_buf_maps(&mut bpf_ingress_ipv6, "EVENTS_IPV6")?;
369    let dropped_packets_ingress_ipv6: PerCpuArray<_, u64> =
370        PerCpuArray::try_from(bpf_ingress_ipv6.take_map("DROPPED_PACKETS").unwrap())?;
371    let matched_packets_ingress_ipv6: PerCpuArray<_, u64> =
372        PerCpuArray::try_from(bpf_ingress_ipv6.take_map("MATCHED_PACKETS").unwrap())?;
373    let submitted_events_ingress_ipv6: PerCpuArray<_, u64> =
374        PerCpuArray::try_from(bpf_ingress_ipv6.take_map("SUBMITTED_EVENTS").unwrap())?;
375    let event_sources_v4;
376    let event_sources_v6;
377    let ebpf_counters;
378
379    if !ingress_only {
380        let mut bpf_egress_ipv4 = load_ebpf_ipv4(interface, TcAttachType::Egress)?;
381        let mut bpf_egress_ipv6 = load_ebpf_ipv6(interface, TcAttachType::Egress)?;
382        let events_egress_ipv4 = take_ring_buf_maps(&mut bpf_egress_ipv4, "EVENTS_IPV4")?;
383        let dropped_packets_egress_ipv4: PerCpuArray<_, u64> =
384            PerCpuArray::try_from(bpf_egress_ipv4.take_map("DROPPED_PACKETS").unwrap())?;
385        let matched_packets_egress_ipv4: PerCpuArray<_, u64> =
386            PerCpuArray::try_from(bpf_egress_ipv4.take_map("MATCHED_PACKETS").unwrap())?;
387        let submitted_events_egress_ipv4: PerCpuArray<_, u64> =
388            PerCpuArray::try_from(bpf_egress_ipv4.take_map("SUBMITTED_EVENTS").unwrap())?;
389        let events_egress_ipv6 = take_ring_buf_maps(&mut bpf_egress_ipv6, "EVENTS_IPV6")?;
390        let dropped_packets_egress_ipv6: PerCpuArray<_, u64> =
391            PerCpuArray::try_from(bpf_egress_ipv6.take_map("DROPPED_PACKETS").unwrap())?;
392        let matched_packets_egress_ipv6: PerCpuArray<_, u64> =
393            PerCpuArray::try_from(bpf_egress_ipv6.take_map("MATCHED_PACKETS").unwrap())?;
394        let submitted_events_egress_ipv6: PerCpuArray<_, u64> =
395            PerCpuArray::try_from(bpf_egress_ipv6.take_map("SUBMITTED_EVENTS").unwrap())?;
396        event_sources_v4 = labeled_ringbuf_sources("egress-ipv4", events_egress_ipv4)
397            .into_iter()
398            .chain(labeled_ringbuf_sources("ingress-ipv4", events_ingress_ipv4))
399            .collect();
400        event_sources_v6 = labeled_ringbuf_sources("egress-ipv6", events_egress_ipv6)
401            .into_iter()
402            .chain(labeled_ringbuf_sources("ingress-ipv6", events_ingress_ipv6))
403            .collect();
404        ebpf_counters = vec![
405            RealtimeEbpfCounters {
406                label: "egress-ipv4",
407                dropped_packets: dropped_packets_egress_ipv4,
408                matched_packets: matched_packets_egress_ipv4,
409                submitted_events: submitted_events_egress_ipv4,
410            },
411            RealtimeEbpfCounters {
412                label: "ingress-ipv4",
413                dropped_packets: dropped_packets_ingress_ipv4,
414                matched_packets: matched_packets_ingress_ipv4,
415                submitted_events: submitted_events_ingress_ipv4,
416            },
417            RealtimeEbpfCounters {
418                label: "egress-ipv6",
419                dropped_packets: dropped_packets_egress_ipv6,
420                matched_packets: matched_packets_egress_ipv6,
421                submitted_events: submitted_events_egress_ipv6,
422            },
423            RealtimeEbpfCounters {
424                label: "ingress-ipv6",
425                dropped_packets: dropped_packets_ingress_ipv6,
426                matched_packets: matched_packets_ingress_ipv6,
427                submitted_events: submitted_events_ingress_ipv6,
428            },
429        ];
430    } else {
431        event_sources_v4 = labeled_ringbuf_sources("ingress-ipv4", events_ingress_ipv4);
432        event_sources_v6 = labeled_ringbuf_sources("ingress-ipv6", events_ingress_ipv6);
433        ebpf_counters = vec![
434            RealtimeEbpfCounters {
435                label: "ingress-ipv4",
436                dropped_packets: dropped_packets_ingress_ipv4,
437                matched_packets: matched_packets_ingress_ipv4,
438                submitted_events: submitted_events_ingress_ipv4,
439            },
440            RealtimeEbpfCounters {
441                label: "ingress-ipv6",
442                dropped_packets: dropped_packets_ingress_ipv6,
443                matched_packets: matched_packets_ingress_ipv6,
444                submitted_events: submitted_events_ingress_ipv6,
445            },
446        ];
447    }
448
449    let enable_source_stats = std::env::var_os("RUSTIFLOW_REALTIME_STATS").is_some();
450    let mut handle_set = JoinSet::new();
451    let (packet_graph, packet_rx) = match packet_graph_mode {
452        PacketGraphMode::Enabled => {
453            let (packet_tx, packet_rx) = watch::channel(Vec::new());
454            let packet_graph = PacketGraphState {
455                packets_since_tick: Arc::new(AtomicU64::new(0)),
456            };
457            handle_set.spawn(packet_graph.clone().publish_intervals(packet_tx));
458            (Some(packet_graph), Some(packet_rx))
459        }
460        PacketGraphMode::Disabled => (None, None),
461    };
462
463    let shard_senders = spawn_shard_workers(
464        shard_count,
465        settings.shard_queue_capacity,
466        &FlowTableSettings {
467            active_timeout,
468            idle_timeout,
469            early_export,
470            expiration_check_interval,
471        },
472        &exporter,
473    );
474
475    // Spawn a drain task and a dispatcher per event source
476    let mut source_stats = Vec::new();
477    let sources = event_sources_v4
478        .into_iter()
479        .map(|(label, ring_buf)| (label, ring_buf, decode_ipv4_event as DecodeEvent))
480        .chain(
481            event_sources_v6
482                .into_iter()
483                .map(|(label, ring_buf)| (label, ring_buf, decode_ipv6_event as DecodeEvent)),
484        );
485    for (label, ring_buf, decode_event) in sources {
486        let stats = enable_source_stats.then(|| Arc::new(RealtimeSourceStats::default()));
487        source_stats.push((label, stats.clone()));
488        let batcher = SourceBatcher::new(
489            &mut handle_set,
490            &shard_senders,
491            &shard_selector,
492            &settings,
493            stats,
494        );
495        handle_set.spawn(drain_ring_source(
496            ring_buf,
497            batcher,
498            packet_graph.clone(),
499            decode_event,
500            realtime_offset_us,
501        ));
502    }
503    // Only the dispatchers keep the shards' senders, so aborting them on
504    // shutdown lets each shard export its remaining flows.
505    drop(shard_senders);
506
507    info!("Waiting for Ctrl-C...");
508
509    if let Some(packet_rx) = packet_rx {
510        let _ = launch_packet_tui(packet_rx).await;
511    }
512
513    signal::ctrl_c().await?;
514
515    // Fetch dropped packets counter from eBPF program before terminating
516    info!("Fetching dropped packet counters before exiting...");
517    let mut total_dropped = 0;
518    for counters in &ebpf_counters {
519        let dropped_packets =
520            read_per_cpu_counter(&counters.dropped_packets, counters.label, "dropped");
521        let matched_packets =
522            read_per_cpu_counter(&counters.matched_packets, counters.label, "matched");
523        let submitted_events =
524            read_per_cpu_counter(&counters.submitted_events, counters.label, "submitted");
525        total_dropped += dropped_packets;
526        info!(
527            "eBPF counters {}: matched_packets={}, submitted_events={}, dropped_packets={}",
528            counters.label, matched_packets, submitted_events, dropped_packets
529        );
530    }
531
532    info!("Total dropped packets before exit: {}", total_dropped);
533    for (label, stats) in &source_stats {
534        if let Some(stats) = stats.as_ref() {
535            log_source_stats(label, stats);
536        }
537    }
538
539    // Cancel the tasks reading ebpf events
540    handle_set.abort_all();
541
542    // Wait for all tasks to finish
543    while let Some(res) = handle_set.join_next().await {
544        match res {
545            Ok(_) => {
546                // Task should never finish by itself
547                error!("Event source task finished unexpectedly");
548            }
549            Err(e) if e.is_cancelled() => {
550                // Task was successfully cancelled
551                debug!("Task was cancelled as part of graceful shutdown");
552            }
553            Err(e) => {
554                // Log other types of errors
555                error!("Task failed: {:?}", e);
556            }
557        }
558    }
559
560    Ok(total_dropped)
561}

563fn read_per_cpu_counter(
564    counter_array: &PerCpuArray<MapData, u64>,
565    label: &str,
566    counter_name: &str,
567) -> u64 {
568    match counter_array.get(&0, 0) {
569        Ok(values) => values.iter().sum(),
570        Err(e) => {
571            error!(
572                "Failed to read {} counter for {}: {:?}",
573                counter_name, label, e
574            );
575            0
576        }
577    }
578}

580fn create_pending_batches(num_shards: usize, shard_batch_size: usize) -> Vec<Vec<PacketFeatures>> {
581    std::iter::repeat_with(|| Vec::with_capacity(shard_batch_size))
582        .take(num_shards)

586async fn enqueue_pending_batches(
587    dispatch_sender: &Sender<ShardDispatchBatch>,
588    pending_batches: &mut [Vec<PacketFeatures>],
589    stats: Option<&Arc<RealtimeSourceStats>>,
590) {
591    for (shard_index, pending_batch) in pending_batches.iter_mut().enumerate() {
592        enqueue_shard_batch(dispatch_sender, pending_batch, stats, shard_index).await;
593    }

598    batch: Vec<PacketFeatures>,

601fn spawn_source_dispatcher(
602    handle_set: &mut JoinSet<()>,
603    shard_senders: Vec<Sender<Vec<PacketFeatures>>>,
604    source_dispatch_queue_capacity: usize,
605    stats: Option<Arc<RealtimeSourceStats>>,
606) -> Sender<ShardDispatchBatch> {
607    let (dispatch_sender, mut dispatch_receiver) =
608        mpsc::channel::<ShardDispatchBatch>(source_dispatch_queue_capacity);
609
610    handle_set.spawn(async move {
611        run_source_dispatcher(&shard_senders, &mut dispatch_receiver, stats.as_ref()).await;
612    });
613
614    dispatch_sender
615}
616
617async fn run_source_dispatcher(
618    shard_senders: &[Sender<Vec<PacketFeatures>>],
619    dispatch_receiver: &mut Receiver<ShardDispatchBatch>,
620    stats: Option<&Arc<RealtimeSourceStats>>,
621) {
622    while let Some(ShardDispatchBatch { shard_index, batch }) = dispatch_receiver.recv().await {
623        let send_start = stats.as_ref().map(|_| Instant::now());
624        if let Err(e) = shard_senders[shard_index].send(batch).await {
625            if let Some(stats) = stats {
626                stats.increment_send_errors();
627            }
628            error!(
629                "Failed to send packet batch to shard {}: {}",
630                shard_index, e
631            );
632        }
633        if let (Some(stats), Some(send_start)) = (stats, send_start) {
634            stats.add_shard_send_wait_ns(elapsed_ns(send_start));
635        }
636    }
637}
638
639async fn enqueue_shard_batch(
640    dispatch_sender: &Sender<ShardDispatchBatch>,
641    pending_batch: &mut Vec<PacketFeatures>,
642    stats: Option<&Arc<RealtimeSourceStats>>,
643    shard_index: usize,
644) {
645    if pending_batch.is_empty() {
646        return;
647    }
648
649    let next_capacity = pending_batch.capacity().max(1);
650    let batch = std::mem::replace(pending_batch, Vec::with_capacity(next_capacity));
651    let send_start = stats.as_ref().map(|_| Instant::now());
652    if let Err(e) = dispatch_sender
653        .send(ShardDispatchBatch { shard_index, batch })
654        .await
655    {
656        if let Some(stats) = stats {
657            stats.increment_send_errors();
658        }
659        error!(
660            "Failed to queue packet batch for shard {}: {}",
661            shard_index, e
662        );
663    }
664    if let (Some(stats), Some(send_start)) = (stats, send_start) {
665        stats.add_dispatch_enqueue_ns(elapsed_ns(send_start));
666    }
667}

677const PACKET_GRAPH_INTERVAL: Duration = Duration::from_millis(100);
678const PACKET_GRAPH_INTERVALS: usize = 100;
679/// Events a source counts locally before adding them to the graph counter, so a
680/// ring that never drains empty still reports progress.
681const PACKET_GRAPH_RECORD_BATCH: u64 = 1024;
682
683/// Packets counted for the realtime graph since its last tick.
684#[derive(Clone)]
685struct PacketGraphState {
686    packets_since_tick: Arc<AtomicU64>,
687}
688
689impl PacketGraphState {
690    /// Sources add once per drained batch, keeping the shared counter off the
691    /// per-packet path.
692    fn record_packets(&self, packets: u64) {
693        self.packets_since_tick
694            .fetch_add(packets, Ordering::Relaxed);
695    }
696
697    /// Publishes per-interval packet counts, newest first, for the last
698    /// `PACKET_GRAPH_INTERVALS` ticks. Intervals without packets count as zero.
699    async fn publish_intervals(self, packet_tx: watch::Sender<Vec<(u64, u64)>>) {
700        let mut intervals = VecDeque::with_capacity(PACKET_GRAPH_INTERVALS + 1);
701        let mut ticker = tokio::time::interval(PACKET_GRAPH_INTERVAL);
702        ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
703        loop {
704            ticker.tick().await;
705            let interval_id = SystemTime::now()
706                .duration_since(UNIX_EPOCH)
707                .map_or(0, |since_epoch| {
708                    (since_epoch.as_millis() / PACKET_GRAPH_INTERVAL.as_millis()) as u64
709                });
710            let packets = self.packets_since_tick.swap(0, Ordering::Relaxed);
711            intervals.push_front((interval_id, packets));
712            intervals.truncate(PACKET_GRAPH_INTERVALS);
713            let _ = packet_tx.send(intervals.iter().copied().collect());
714        }
715    }
716}

718fn compute_realtime_offset_us() -> Result<i64, anyhow::Error> {
719    let realtime_us = read_clock_us(libc::CLOCK_REALTIME)?;
720    let monotonic_us = read_clock_us(libc::CLOCK_MONOTONIC)?;
721    Ok(realtime_us - monotonic_us)
722}

739fn take_ring_buf_maps(
740    bpf: &mut Ebpf,
741    base_name: &str,

757/// Forwards `aya-log` records from the eBPF program to the userspace logger.
758fn spawn_ebpf_logger(bpf: &mut Ebpf) {
759    let logger = match EbpfLogger::init(bpf) {
760        Ok(logger) => logger,
761        Err(e) => {
762            debug!("eBPF logger unavailable: {e}");
763            return;
764        }
765    };
766    // SAFETY: the logger owns its ring buffer map file descriptor and reports
767    // that same descriptor from `as_raw_fd` until it is dropped with the `AsyncFd`.
768    let mut logger = match unsafe { AsyncFd::register(logger) } {
769        Ok(logger) => logger,
770        Err(e) => {
771            error!("Failed to poll the eBPF logger: {e}");
772            return;
773        }
774    };
775
776    tokio::spawn(async move {
777        loop {
778            let Ok(mut guard) = logger.readable_mut().await else {
779                return;
780            };
781            guard.get_inner_mut().flush();
782            guard.clear_ready();
783        }
784    });
785}

787fn read_clock_us(clock_id: libc::clockid_t) -> Result<i64, anyhow::Error> {
788    let mut ts = libc::timespec {
789        tv_sec: 0,
790        tv_nsec: 0,
791    };
792
793    if unsafe { libc::clock_gettime(clock_id, &mut ts) } != 0 {
794        return Err(io::Error::last_os_error().into());
795    }
796
797    Ok(ts.tv_sec * 1_000_000 + ts.tv_nsec / 1_000)
798}

800fn bump_memlock_rlimit() {
801    // Bump the memlock rlimit. This is needed for older kernels that don't use the
802    // new memcg based accounting, see https://lwn.net/Articles/837122/
803    let rlim = libc::rlimit {
804        rlim_cur: libc::RLIM_INFINITY,
805        rlim_max: libc::RLIM_INFINITY,
806    };
807    let ret = unsafe { libc::setrlimit(libc::RLIMIT_MEMLOCK, &rlim) };
808    if ret != 0 {
809        debug!("remove limit on locked memory failed, ret is: {}", ret);
810    }
811}

837fn load_ebpf_ipv4(interface: &str, tc_attach_type: TcAttachType) -> Result<Ebpf, anyhow::Error> {
838    let binary_path = ebpf_binary_path("rustiflow-ebpf-ipv4");
839    let attach_label = tc_attach_type_label(tc_attach_type);
840    info!(
841        "Loading IPv4 eBPF binary {} for {} on {}",
842        binary_path.display(),
843        attach_label,
844        interface
845    );
846    let mut bpf_ipv4 = Ebpf::load_file(&binary_path).with_context(|| {
847        format!(
848            "Failed to load eBPF IPv4 binary from {}. Build it first with `cargo xtask ebpf-ipv4`.",
849            binary_path.display()
850        )
851    })?;
852
853    // Attach the eBPF program function
854    spawn_ebpf_logger(&mut bpf_ipv4);
855    match tc::qdisc_add_clsact(interface) {
856        Ok(()) => info!("Ensured clsact qdisc on {}", interface),
857        Err(e) => debug!("qdisc_add_clsact({}): {:?}", interface, e),
858    }
859
860    let program_egress_ipv4: &mut SchedClassifier =
861        bpf_ipv4.program_mut("tc_flow_track").unwrap().try_into()?;
862    info!(
863        "Loading IPv4 tc classifier for {} on {}",
864        attach_label, interface
865    );
866    program_egress_ipv4.load().map_err(|e| {
867        error!("Failed to load eBPF program: {:?}", e);
868        e
869    })?;
870    info!(
871        "Attaching IPv4 tc classifier to {} on {}",
872        attach_label, interface
873    );
874    program_egress_ipv4
875        .attach_with_options(
876            interface,
877            tc_attach_type,
878            TcAttachOptions::Netlink(NlOptions::default()),
879        )
880        .map_err(|e| {
881            error!("Failed to attach eBPF program: {:?}", e);
882            e
883        })?;
884    info!(
885        "Attached IPv4 tc classifier to {} on {}",
886        attach_label, interface
887    );
888
889    Ok(bpf_ipv4)
890}
