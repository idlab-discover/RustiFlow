53struct RealtimeSourceStats {
54    ring: Option<RingStats>,
55    events: AtomicU64,
56    ring_age_us_log2: [AtomicU64; 32],
57    decode_and_shard_ns: AtomicU64,
58    dispatch_enqueue_ns: AtomicU64,
59    shard_send_wait_ns: AtomicU64,
60    total_event_ns: AtomicU64,
61    send_errors: AtomicU64,
62}

64impl RealtimeSourceStats {
65    fn add_decode_and_shard_ns(&self, value: u64) {
66        self.decode_and_shard_ns.fetch_add(value, Ordering::Relaxed);
67    }
68
69    fn add_dispatch_enqueue_ns(&self, value: u64) {
70        self.dispatch_enqueue_ns.fetch_add(value, Ordering::Relaxed);
71    }
72
73    fn add_shard_send_wait_ns(&self, value: u64) {
74        self.shard_send_wait_ns.fetch_add(value, Ordering::Relaxed);
75    }
76
77    fn add_total_event_ns(&self, value: u64) {
78        self.total_event_ns.fetch_add(value, Ordering::Relaxed);
79    }
80
81    fn sample_ring_age(&self, event: &[u8]) {
82        if !self.events.load(Ordering::Relaxed).is_multiple_of(64) {
83            return;
84        }
85        // Both event ABIs start with a native-endian CLOCK_MONOTONIC timestamp.
86        let Some(timestamp) = event.first_chunk::<8>() else {
87            return;
88        };
89        if let Ok(now_us) = read_clock_us(libc::CLOCK_MONOTONIC) {
90            let age_us = (now_us as u64).saturating_sub(u64::from_ne_bytes(*timestamp) / 1_000);
91            let bucket = age_us.max(1).ilog2().min(31) as usize;
92            self.ring_age_us_log2[bucket].fetch_add(1, Ordering::Relaxed);
93        }
94    }
95
96    fn increment_events(&self) {
97        self.events.fetch_add(1, Ordering::Relaxed);
98    }
99
100    fn increment_send_errors(&self) {
101        self.send_errors.fetch_add(1, Ordering::Relaxed);
102    }
103}

109fn log_source_stats(label: &str, stats: &RealtimeSourceStats) {
110    if let Some(ring) = &stats.ring {
111        ring.log(label);
112    }
113    let events = stats.events.load(Ordering::Relaxed);
114    let decode_and_shard_ns = stats.decode_and_shard_ns.load(Ordering::Relaxed);
115    let dispatch_enqueue_ns = stats.dispatch_enqueue_ns.load(Ordering::Relaxed);
116    let shard_send_wait_ns = stats.shard_send_wait_ns.load(Ordering::Relaxed);
117    let total_event_ns = stats.total_event_ns.load(Ordering::Relaxed);
118    let send_errors = stats.send_errors.load(Ordering::Relaxed);
119
120    if events == 0 {
121        info!("Realtime source {}: no events drained", label);
122        return;
123    }
124
125    info!(
126        "Realtime source {}: events={} total_ms={:.3} decode_ms={:.3} enqueue_wait_ms={:.3} shard_send_wait_ms={:.3} avg_event_us={:.3} avg_enqueue_wait_us={:.3} avg_shard_send_wait_us={:.3} send_errors={}",
127        label,
128        events,
129        total_event_ns as f64 / 1_000_000.0,
130        decode_and_shard_ns as f64 / 1_000_000.0,
131        dispatch_enqueue_ns as f64 / 1_000_000.0,
132        shard_send_wait_ns as f64 / 1_000_000.0,
133        total_event_ns as f64 / events as f64 / 1_000.0,
134        dispatch_enqueue_ns as f64 / events as f64 / 1_000.0,
135        shard_send_wait_ns as f64 / events as f64 / 1_000.0,
136        send_errors,
137    );
138    let ring_age = stats
139        .ring_age_us_log2
140        .each_ref()
141        .map(|bucket| bucket.load(Ordering::Relaxed));
142    if ring_age.iter().any(|&count| count != 0) {
143        info!("Realtime source {label}: ring_age_us_log2={ring_age:?}");
144    }
145}

202/// Spawns one flow-table worker per shard. Each worker exports its remaining
203/// flows once every sender for its shard is dropped.
204fn spawn_shard_workers<T: Flow>(
205    shard_count: NonZeroUsize,
206    settings: &PipelineSettings,
207    flow_table_settings: &FlowTableSettings,
208    exporter: &impl FlowSink<T>,
209) -> Vec<ShardBatchSender> {
210    debug!("Creating {} sharded FlowTables...", shard_count);
211    let shard_senders = (0..shard_count.get())
212        .map(|_| {
213            let (sender, mut rx) =
214                mpsc::channel::<Vec<PacketFeatures>>(settings.shard_queue_capacity);
215            let batch_pool = settings
216                .recycle_shard_batches
217                .then(|| Arc::new(BatchPool::new()));
218            let worker_pool = batch_pool.clone();
219            let mut canonicalizer = BatchCanonicalizer::new(settings.simd_keys);
220            let mut flow_table = FlowTable::new(
221                flow_table_settings.active_timeout,
222                flow_table_settings.idle_timeout,
223                flow_table_settings.early_export,
224                exporter.shard_output(),
225                flow_table_settings.expiration_check_interval,
226            );
227
228            tokio::spawn(async move {
229                let mut last_timestamp = None;
230                while let Some(packet_batch) = rx.recv().await {
231                    flow_table
232                        .process_batch(&packet_batch, &mut canonicalizer)
233                        .await;
234                    if let Some(packet) = packet_batch.last() {
235                        last_timestamp = Some(packet.timestamp_us);
236                    }
237                    if let Some(pool) = &worker_pool {
238                        pool.recycle(packet_batch);
239                    }
240                }
241                debug!("Shard finished processing packets");
242                // Handle flow exporting when the receiver is closed
243                if let Some(timestamp) = last_timestamp {
244                    flow_table.export_all_flows(timestamp).await;
245                }
246            });
247            ShardBatchSender { sender, batch_pool }
248        })
249        .collect();
250    debug!("Sharded FlowTables created");
251    shard_senders
252}

273/// One event source's userspace stage: pick a shard per packet, batch per
274/// shard, and hand batches to the source's own dispatcher task so draining
275/// never waits on a shard queue directly.
276struct SourceBatcher {
277    shard_selector: ShardSelector,
278    shard_batch_size: usize,
279    pending_batches: Vec<PendingShardBatch>,
280    dispatch_sender: Sender<ShardDispatchBatch>,
281    stats: Option<Arc<RealtimeSourceStats>>,
282}
283
284struct PendingShardBatch {
285    packets: Vec<PacketFeatures>,
286    batch_pool: Option<Arc<BatchPool>>,
287}
288
289impl SourceBatcher {
290    /// Creates the batcher and spawns its dispatcher into `handle_set`.
291    fn new(
292        handle_set: &mut JoinSet<anyhow::Result<()>>,
293        shard_senders: &[ShardBatchSender],
294        shard_selector: &ShardSelector,
295        settings: &PipelineSettings,
296        stats: Option<Arc<RealtimeSourceStats>>,
297    ) -> Self {
298        let dispatch_sender = spawn_source_dispatcher(
299            handle_set,
300            shard_senders
301                .iter()
302                .map(|shard| shard.sender.clone())
303                .collect(),
304            settings.source_dispatch_queue_capacity,
305            stats.clone(),
306        );
307        Self {
308            shard_selector: shard_selector.clone(),
309            shard_batch_size: settings.shard_batch_size,
310            pending_batches: shard_senders
311                .iter()
312                .map(|shard| PendingShardBatch {
313                    packets: Vec::with_capacity(settings.shard_batch_size),
314                    batch_pool: shard.batch_pool.clone(),
315                })
316                .collect(),
317            dispatch_sender,
318            stats,
319        }
320    }
321
322    /// Decodes one event and queues it for its shard, dispatching the shard's
323    /// batch once it is full.
324    async fn push_event(&mut self, decode: impl FnOnce() -> Option<PacketFeatures>) -> bool {
325        let event_start = self.stats.is_some().then(Instant::now);
326        let Some(packet_features) = decode() else {
327            return false;
328        };
329        let shard_index = self
330            .shard_selector
331            .shard_index(&packet_features.biflow_key_value());
332        if let (Some(stats), Some(event_start)) = (&self.stats, event_start) {
333            stats.add_decode_and_shard_ns(elapsed_ns(event_start));
334        }
335
336        let pending_batch = &mut self.pending_batches[shard_index];
337        pending_batch.packets.push(packet_features);
338        if pending_batch.packets.len() >= self.shard_batch_size {
339            enqueue_shard_batch(
340                &self.dispatch_sender,
341                pending_batch,
342                self.stats.as_ref(),
343                shard_index,
344            )
345            .await;
346        }
347
348        if let (Some(stats), Some(event_start)) = (&self.stats, event_start) {
349            stats.increment_events();
350            stats.add_total_event_ns(elapsed_ns(event_start));
351        }
352        true
353    }
354
355    /// Dispatches every partial batch, as a source does when its ring runs empty.
356    async fn flush(&mut self) {
357        for (shard_index, pending_batch) in self.pending_batches.iter_mut().enumerate() {
358            enqueue_shard_batch(
359                &self.dispatch_sender,
360                pending_batch,
361                self.stats.as_ref(),
362                shard_index,
363            )
364            .await;
365        }
366    }
367}

369/// Starts the realtime processing of packets on the given interface.
370/// The function will return the number of packets dropped by the eBPF program.
371#[allow(clippy::too_many_arguments)]
372pub async fn handle_realtime<T>(
373    interface: &str,
374    exporter: impl FlowSink<T>,
375    num_threads: u8,
376    active_timeout: u64,
377    idle_timeout: u64,
378    early_export: Option<NonZeroU64>,
379    expiration_check_interval: u64,
380    ingress_only: bool,
381    promiscuous_mode: PromiscuousMode,
382    packet_graph_mode: PacketGraphMode,
383) -> Result<u64, anyhow::Error>
384where
385    T: Flow,
386{
387    let reactor_count = ReactorCount::from_env(ingress_only)?;
388    let settings = PipelineSettings::from_env(PipelineMode::Capture);
389    let shard_count =
390        NonZeroUsize::new(usize::from(num_threads)).context("thread count must be at least 1")?;
391    let shard_selector = ShardSelector::new(shard_count);
392
393    // Needed for older kernels
394    bump_memlock_rlimit();
395
396    let realtime_offset_us = compute_realtime_offset_us()?;
397
398    // Join before attaching and declare first, so the membership is dropped
399    // only after every classifier has been detached.
400    let _promiscuous_membership = match promiscuous_mode {
401        PromiscuousMode::Enabled => {
402            let membership = PromiscuousMembership::join(interface)?;
403            info!("Holding {interface} in promiscuous mode until capture stops");
404            Some(membership)
405        }
406        PromiscuousMode::Disabled => None,
407    };
408
409    let mut capture = Capture::attach(interface, ingress_only)?;
410    let (event_sources_v4, event_sources_v6) = capture.take_sources();
411
412    let enable_source_stats = std::env::var_os("RUSTIFLOW_REALTIME_STATS").is_some();
413    let mut handle_set = JoinSet::new();
414    let mut graph_tasks = JoinSet::new();
415    let (packet_graph, packet_rx) = match packet_graph_mode {
416        PacketGraphMode::Enabled => {
417            let (packet_tx, packet_rx) = watch::channel(Vec::new());
418            let packet_graph = PacketGraphState {
419                packets_since_tick: Arc::new(AtomicU64::new(0)),
420            };
421            graph_tasks.spawn(packet_graph.clone().publish_intervals(packet_tx));
422            (Some(packet_graph), Some(packet_rx))
423        }
424        PacketGraphMode::Disabled => (None, None),
425    };
426
427    let shard_senders = spawn_shard_workers(
428        shard_count,
429        &settings,
430        &FlowTableSettings {
431            active_timeout,
432            idle_timeout,
433            early_export,
434            expiration_check_interval,
435        },
436        &exporter,
437    );
438
439    // Spawn a drain task and a dispatcher per event source
440    let drain_policy = DrainPolicy::from_env();
441    let mut source_stats = Vec::new();
442    let sources = event_sources_v4
443        .into_iter()
444        .map(|(label, ring_buf)| (label, ring_buf, decode_ipv4_event as DecodeEvent))
445        .chain(
446            event_sources_v6
447                .into_iter()
448                .map(|(label, ring_buf)| (label, ring_buf, decode_ipv6_event as DecodeEvent)),
449        );
450    let sources = sources
451        .map(|(label, ring, decode)| {
452            let stats = enable_source_stats.then(|| {
453                Arc::new(RealtimeSourceStats {
454                    ring: Some(RingStats::new(realtime_offset_us)),
455                    ..RealtimeSourceStats::default()
456                })
457            });
458
459            source_stats.push((label, stats.clone()));
460            RingSource {
461                ring,
462                decode,
463                stats,
464            }
465        })
466        .collect();
467    let mut reactors = RingSourcePipeline {
468        shard_senders,
469        shard_selector,
470        settings,
471        packet_graph,
472        drain_policy,
473        realtime_offset_us,
474    }
475    .start(reactor_count, &mut handle_set, sources)
476    .await?;
477
478    info!("Waiting for Ctrl-C...");
479
480    if let Some(packet_rx) = packet_rx {
481        let _ = launch_packet_tui(packet_rx).await;
482    }
483
484    let capture_result = tokio::select! {
485        signal = signal::ctrl_c() => signal.context("waiting for Ctrl-C"),
486        error = reactors.unexpected_exit() => Err(error),
487        result = handle_set.join_next(), if !handle_set.is_empty() => {
488            Err(unexpected_source_exit(result))
489        }
490    };
491    graph_tasks.shutdown().await;
492
493    // Detach first so the source tasks can finish draining finite rings.
494    let detach_result = capture.detach();
495    let shutdown = if detach_result.is_ok() {
496        Shutdown::Drain
497    } else {
498        Shutdown::Abort
499    };
500    let drain_result = reactors.shutdown(shutdown, &mut handle_set).await;
501    for (label, stats) in &source_stats {
502        if let Some(stats) = stats.as_ref() {
503            log_source_stats(label, stats);
504        }
505    }
506
507    capture_result?;
508    detach_result?;
509    let decoded = drain_result?;
510    let totals = capture.log_totals()?;
511    info!("Total dropped packets before exit: {}", totals.dropped);
512    anyhow::ensure!(
513        decoded == totals.submitted,
514        "realtime capture submitted {} events but decoded {decoded}",
515        totals.submitted,
516    );
517    Ok(totals.dropped)
518}

525fn spawn_source_dispatcher(
526    handle_set: &mut JoinSet<anyhow::Result<()>>,
527    shard_senders: Vec<Sender<Vec<PacketFeatures>>>,
528    source_dispatch_queue_capacity: usize,
529    stats: Option<Arc<RealtimeSourceStats>>,
530) -> Sender<ShardDispatchBatch> {
531    let (dispatch_sender, mut dispatch_receiver) =
532        mpsc::channel::<ShardDispatchBatch>(source_dispatch_queue_capacity);
533
534    handle_set.spawn(async move {
535        run_source_dispatcher(&shard_senders, &mut dispatch_receiver, stats.as_ref()).await
536    });
537
538    dispatch_sender
539}
540
541async fn run_source_dispatcher(
542    shard_senders: &[Sender<Vec<PacketFeatures>>],
543    dispatch_receiver: &mut Receiver<ShardDispatchBatch>,
544    stats: Option<&Arc<RealtimeSourceStats>>,
545) -> anyhow::Result<()> {
546    while let Some(ShardDispatchBatch { shard_index, batch }) = dispatch_receiver.recv().await {
547        let handoff = stats
548            .and_then(|stats| stats.ring.as_ref())
549            .zip(batch.first())
550            .map(|(ring, first)| (ring, batch.len(), first.timestamp_us));
551        let send_start = stats.as_ref().map(|_| Instant::now());
552        if shard_senders[shard_index].send(batch).await.is_err() {
553            if let Some(stats) = stats {
554                stats.increment_send_errors();
555            }
556            anyhow::bail!("packet queue for shard {shard_index} closed during dispatch");
557        } else if let Some((ring, records, first_timestamp_us)) = handoff {
558            ring.record_handoff(records, first_timestamp_us);
559        }
560        if let (Some(stats), Some(send_start)) = (stats, send_start) {
561            stats.add_shard_send_wait_ns(elapsed_ns(send_start));
562        }
563    }
564    Ok(())
565}
566
567async fn enqueue_shard_batch(
568    dispatch_sender: &Sender<ShardDispatchBatch>,
569    pending_batch: &mut PendingShardBatch,
570    stats: Option<&Arc<RealtimeSourceStats>>,
571    shard_index: usize,
572) {
573    if pending_batch.packets.is_empty() {
574        return;
575    }
576
577    let replacement = pending_batch
578        .batch_pool
579        .as_ref()
580        .and_then(|pool| pool.try_take())
581        .unwrap_or_else(|| Vec::with_capacity(pending_batch.packets.capacity()));
582    let batch = std::mem::replace(&mut pending_batch.packets, replacement);
583    let send_start = stats.as_ref().map(|_| Instant::now());
584    if let Err(e) = dispatch_sender
585        .send(ShardDispatchBatch { shard_index, batch })
586        .await
587    {
588        if let Some(stats) = stats {
589            stats.increment_send_errors();
590        }
591        error!(
592            "Failed to queue packet batch for shard {}: {}",
593            shard_index, e
594        );
595    }
596    if let (Some(stats), Some(send_start)) = (stats, send_start) {
597        stats.add_dispatch_enqueue_ns(elapsed_ns(send_start));
598    }
599}

601fn read_env_usize(var_name: &str, default_value: usize) -> usize {
602    std::env::var(var_name)
603        .ok()
604        .and_then(|value| value.parse::<usize>().ok())

606        .unwrap_or(default_value)
607}

609const PACKET_GRAPH_INTERVAL: Duration = Duration::from_millis(100);
610const PACKET_GRAPH_INTERVALS: usize = 100;
611/// Events a source counts locally before adding them to the graph counter, so a
612/// ring that never drains empty still reports progress.
613const PACKET_GRAPH_RECORD_BATCH: u64 = 1024;
614
615/// Packets counted for the realtime graph since its last tick.
616#[derive(Clone)]
617struct PacketGraphState {
618    packets_since_tick: Arc<AtomicU64>,
619}
620
621impl PacketGraphState {
622    /// Sources add once per drained batch, keeping the shared counter off the
623    /// per-packet path.
624    fn record_packets(&self, packets: u64) {
625        self.packets_since_tick
626            .fetch_add(packets, Ordering::Relaxed);
627    }
628
629    /// Publishes per-interval packet counts, newest first, for the last
630    /// `PACKET_GRAPH_INTERVALS` ticks. Intervals without packets count as zero.
631    async fn publish_intervals(self, packet_tx: watch::Sender<Vec<(u64, u64)>>) {
632        let mut intervals = VecDeque::with_capacity(PACKET_GRAPH_INTERVALS + 1);
633        let mut ticker = tokio::time::interval(PACKET_GRAPH_INTERVAL);
634        ticker.set_missed_tick_behavior(MissedTickBehavior::Delay);
635        loop {
636            ticker.tick().await;
637            let interval_id = SystemTime::now()
638                .duration_since(UNIX_EPOCH)
639                .map_or(0, |since_epoch| {
640                    (since_epoch.as_millis() / PACKET_GRAPH_INTERVAL.as_millis()) as u64
641                });
642            let packets = self.packets_since_tick.swap(0, Ordering::Relaxed);
643            intervals.push_front((interval_id, packets));
644            intervals.truncate(PACKET_GRAPH_INTERVALS);
645            let _ = packet_tx.send(intervals.iter().copied().collect());
646        }
647    }
648}

650fn compute_realtime_offset_us() -> Result<i64, anyhow::Error> {
651    let realtime_us = read_clock_us(libc::CLOCK_REALTIME)?;
652    let monotonic_us = read_clock_us(libc::CLOCK_MONOTONIC)?;
653    Ok(realtime_us - monotonic_us)
654}

660fn read_clock_us(clock_id: libc::clockid_t) -> Result<i64, anyhow::Error> {
661    let mut ts = libc::timespec {
662        tv_sec: 0,
663        tv_nsec: 0,
664    };
665
666    if unsafe { libc::clock_gettime(clock_id, &mut ts) } != 0 {
667        return Err(io::Error::last_os_error().into());
668    }
669
670    Ok(ts.tv_sec * 1_000_000 + ts.tv_nsec / 1_000)
671}

673fn bump_memlock_rlimit() {
674    // Bump the memlock rlimit. This is needed for older kernels that don't use the
675    // new memcg based accounting, see https://lwn.net/Articles/837122/
676    let rlim = libc::rlimit {
677        rlim_cur: libc::RLIM_INFINITY,
678        rlim_max: libc::RLIM_INFINITY,
679    };
680    let ret = unsafe { libc::setrlimit(libc::RLIMIT_MEMLOCK, &rlim) };
681    if ret != 0 {
682        debug!("remove limit on locked memory failed, ret is: {}", ret);
683    }
684}
